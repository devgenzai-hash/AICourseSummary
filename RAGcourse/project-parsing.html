P2: Document Parsing — All 9 Types

Parse every document type the enterprise has: PDFs, Word, emails, web pages, databases, Slack, code repos, scanned papers, and bills. One unified output format.

Advanced Full Code Production

Unified Document Schema

Every parser — regardless of input format — produces the same output structure. This is critical for the downstream pipeline.

# src/parsers/base.py

from dataclasses import dataclass, field
from datetime import datetime
from typing import List, Dict, Optional
from enum import Enum
from abc import ABC, abstractmethod
from pathlib import Path

class SourceType(Enum):
    PDF = "pdf"
    DOCX = "docx"
    EMAIL = "email"
    WEB = "web"
    DATABASE = "database"
    SLACK = "slack"
    CODE = "code"
    OCR = "ocr"
    BILL = "bill"

@dataclass
class ParsedDocument:
    """Universal document output from any parser."""
    id: str                          # Unique identifier
    content: str                     # Full text content (cleaned)
    source_type: SourceType          # Which parser produced it
    source_path: str                 # Original file path or URL
    title: Optional[str] = None     # Document title
    author: Optional[str] = None    # Author/sender
    created_at: Optional[datetime] = None
    updated_at: Optional[datetime] = None
    language: str = "en"
    page_count: Optional[int] = None
    tables: List[Dict] = field(default_factory=list)  # Extracted tables
    metadata: Dict = field(default_factory=dict)     # Extra metadata
    sections: List[Dict] = field(default_factory=list) # Structural sections
    confidence: float = 1.0          # Parse confidence (OCR may be lower)

class BaseParser(ABC):
    """Abstract base class for all document parsers."""
    
    @abstractmethod
    def parse(self, source: str) -> List[ParsedDocument]:
        """Parse a source into one or more ParsedDocument objects."""
        ...
    
    @abstractmethod
    def can_handle(self, source: str) -> bool:
        """Return True if this parser can handle the given source."""
        ...
    
    def _generate_id(self, source: str, content: str) -> str:
        """Generate a deterministic document ID."""
        import hashlib
        raw = f"{source}:{content[:500]}"
        return hashlib.sha256(raw.encode()).hexdigest()[:16]

import mimetypes
from pathlib import Path
from typing import List
from .base import BaseParser, ParsedDocument

class DocumentRouter:
    """
    Auto-detects document type and routes to the correct parser.
    Handles: PDF, DOCX, Email, Web, DB, Slack, Code, OCR, Bills.
    """
    
    def __init__(self):
        # Lazy imports to avoid loading all parsers at startup
        from .pdf_parser import PDFParser
        from .docx_parser import DocxParser
        from .email_parser import EmailParser
        from .web_parser import WebParser
        from .db_parser import DatabaseParser
        from .slack_parser import SlackParser
        from .code_parser import CodeParser
        from .ocr_parser import OCRParser
        from .bill_parser import BillParser
        
        self.parsers: List[BaseParser] = [
            PDFParser(),
            DocxParser(),
            EmailParser(),
            WebParser(),
            DatabaseParser(),
            SlackParser(),
            CodeParser(),
            BillParser(),    # Before OCR (bills are structured PDFs)
            OCRParser(),     # Fallback for image-based docs
        ]
    
    def parse(self, source: str) -> List[ParsedDocument]:
        """
        Route a source to the appropriate parser.
        
        Args:
            source: File path, URL, or connection string
        
        Returns:
            List of ParsedDocument objects
        """
        for parser in self.parsers:
            if parser.can_handle(source):
                try:
                    docs = parser.parse(source)
                    print(f"✅ Parsed {source} → {len(docs)} docs via {parser.__class__.__name__}")
                    return docs
                except Exception as e:
                    print(f"⚠️ {parser.__class__.__name__} failed on {source}: {e}")
                    continue
        
        raise ValueError(f"No parser found for: {source}")
    
    def parse_directory(self, directory: str) -> List[ParsedDocument]:
        """Parse all files in a directory recursively."""
        all_docs = []
        dir_path = Path(directory)
        
        for file_path in dir_path.rglob("*"):
            if file_path.is_file():
                try:
                    docs = self.parse(str(file_path))
                    all_docs.extend(docs)
                except ValueError:
                    print(f"⏭️ Skipping unsupported file: {file_path}")
        
        print(f"\n📊 Total: Parsed {len(all_docs)} documents from {directory}")
        return all_docs

1. PDF Parser (Company Reports, Research Papers, Manuals)

💡 Why pymupdf4llm?

pymupdf4llm converts PDFs to markdown format — preserving headings, tables, bold/italic, and page structure. Perfect for LLM consumption. 10x faster than pdfplumber for large documents.

# src/parsers/pdf_parser.py

import pymupdf4llm
import fitz  # PyMuPDF
from pathlib import Path
from datetime import datetime
from typing import List
from .base import BaseParser, ParsedDocument, SourceType

class PDFParser(BaseParser):
    """
    Parse PDF documents: reports, papers, manuals, contracts.
    Uses pymupdf4llm for markdown extraction with structure preservation.
    Falls back to OCR for image-based PDFs.
    """
    
    def can_handle(self, source: str) -> bool:
        return Path(source).suffix.lower() == ".pdf"
    
    def parse(self, source: str) -> List[ParsedDocument]:
        path = Path(source)
        doc = fitz.open(source)
        
        # Extract metadata from PDF
        meta = doc.metadata
        
        # Check if PDF is image-based (scanned)
        if self._is_scanned_pdf(doc):
            # Delegate to OCR parser
            return []  # Router will fall through to OCRParser
        
        # Extract content as markdown (preserves structure)
        md_content = pymupdf4llm.to_markdown(
            source,
            page_chunks=True,    # Split by page for large docs
            write_images=False,  # Skip embedded images
            show_progress=False
        )
        
        # pymupdf4llm returns list of page dicts when page_chunks=True
        full_text = "\n\n".join(
            page["text"] for page in md_content
        )
        
        # Extract tables separately for structured data
        tables = self._extract_tables(doc)
        
        # Build sections from markdown headings
        sections = self._extract_sections(full_text)
        
        parsed = ParsedDocument(
            id=self._generate_id(source, full_text),
            content=full_text,
            source_type=SourceType.PDF,
            source_path=str(path.absolute()),
            title=meta.get("title") or path.stem,
            author=meta.get("author"),
            created_at=self._parse_pdf_date(meta.get("creationDate")),
            updated_at=self._parse_pdf_date(meta.get("modDate")),
            page_count=len(doc),
            tables=tables,
            sections=sections,
            metadata={
                "producer": meta.get("producer", ""),
                "subject": meta.get("subject", ""),
                "keywords": meta.get("keywords", ""),
                "file_size_mb": path.stat().st_size / (1024 * 1024)
            }
        )
        
        doc.close()
        return [parsed]
    
    def _is_scanned_pdf(self, doc) -> bool:
        """Detect if PDF is image-based (no extractable text)."""
        # Check first 3 pages for text
        for page_num in range(min(3, len(doc))):
            text = doc[page_num].get_text().strip()
            if len(text) > 50:
                return False
        return True
    
    def _extract_tables(self, doc) -> List[Dict]:
        """Extract tables from PDF pages."""
        tables = []
        for page_num in range(len(doc)):
            page = doc[page_num]
            page_tables = page.find_tables()
            for table in page_tables:
                tables.append({
                    "page": page_num + 1,
                    "data": table.extract(),
                    "headers": table.header.names if table.header else []
                })
        return tables
    
    def _extract_sections(self, markdown: str) -> List[Dict]:
        """Parse markdown headings into sections."""
        sections = []
        current_section = None
        
        for line in markdown.split("\n"):
            if line.startswith("# "):
                if current_section:
                    sections.append(current_section)
                current_section = {"heading": line[2:], "level": 1, "content": ""}
            elif line.startswith("## "):
                if current_section:
                    sections.append(current_section)
                current_section = {"heading": line[3:], "level": 2, "content": ""}
            elif current_section:
                current_section["content"] += line + "\n"
        
        if current_section:
            sections.append(current_section)
        return sections
    
    def _parse_pdf_date(self, date_str) -> Optional[datetime]:
        """Parse PDF date format (D:20240101120000)."""
        if not date_str:
            return None
        try:
            clean = date_str.replace("D:", "")[:14]
            return datetime.strptime(clean, "%Y%m%d%H%M%S")
        except:
            return None

2. Word/DOCX Parser (Memos, Proposals, Meeting Notes)

# src/parsers/docx_parser.py

from docx import Document
from docx.enum.text import WD_PARAGRAPH_ALIGNMENT
from pathlib import Path
from datetime import datetime
from typing import List, Dict
from .base import BaseParser, ParsedDocument, SourceType

class DocxParser(BaseParser):
    """
    Parse Word documents: memos, proposals, meeting notes.
    Preserves headings, tables, and document structure.
    """
    
    def can_handle(self, source: str) -> bool:
        return Path(source).suffix.lower() in (".docx", ".doc")
    
    def parse(self, source: str) -> List[ParsedDocument]:
        path = Path(source)
        doc = Document(source)
        
        # Extract all paragraphs with structure
        content_parts = []
        sections = []
        current_section = {"heading": "Introduction", "level": 1, "content": ""}
        
        for para in doc.paragraphs:
            if para.style.name.startswith("Heading"):
                # Save previous section
                if current_section["content"].strip():
                    sections.append(current_section)
                
                level = int(para.style.name.replace("Heading ", "") or "1")
                current_section = {
                    "heading": para.text,
                    "level": level,
                    "content": ""
                }
                content_parts.append(f"\n{'#' * level} {para.text}\n")
            else:
                content_parts.append(para.text)
                current_section["content"] += para.text + "\n"
        
        if current_section["content"].strip():
            sections.append(current_section)
        
        # Extract tables
        tables = []
        for table in doc.tables:
            rows = []
            for row in table.rows:
                rows.append([cell.text for cell in row.cells])
            tables.append({
                "headers": rows[0] if rows else [],
                "data": rows[1:] if len(rows) > 1 else rows
            })
        
        # Get core properties
        props = doc.core_properties
        full_text = "\n".join(content_parts)
        
        return [ParsedDocument(
            id=self._generate_id(source, full_text),
            content=full_text,
            source_type=SourceType.DOCX,
            source_path=str(path.absolute()),
            title=props.title or path.stem,
            author=props.author,
            created_at=props.created,
            updated_at=props.modified,
            tables=tables,
            sections=sections,
            metadata={
                "category": props.category or "",
                "comments": props.comments or "",
                "revision": props.revision
            }
        )]

3. Email Parser (Customer Conversations, Internal Communications)

# src/parsers/email_parser.py

import email
from email import policy
from pathlib import Path
from datetime import datetime
from typing import List
import extract_msg
from .base import BaseParser, ParsedDocument, SourceType

class EmailParser(BaseParser):
    """
    Parse email files: .eml (standard) and .msg (Outlook).
    Extracts subject, sender, recipients, body, and attachments list.
    """
    
    def can_handle(self, source: str) -> bool:
        return Path(source).suffix.lower() in (".eml", ".msg")
    
    def parse(self, source: str) -> List[ParsedDocument]:
        path = Path(source)
        
        if path.suffix.lower() == ".msg":
            return self._parse_msg(path)
        else:
            return self._parse_eml(path)
    
    def _parse_eml(self, path: Path) -> List[ParsedDocument]:
        """Parse standard .eml email files."""
        with open(path, "rb") as f:
            msg = email.message_from_binary_file(f, policy=policy.default)
        
        # Extract body text
        body = msg.get_body(preferencelist=("plain", "html"))
        content = body.get_content() if body else ""
        
        # Clean HTML if needed
        if "<" in content and ">" in content:
            from bs4 import BeautifulSoup
            content = BeautifulSoup(content, "html.parser").get_text(separator="\n")
        
        # Get attachment names
        attachments = [
            part.get_filename()
            for part in msg.walk()
            if part.get_content_disposition() == "attachment"
        ]
        
        # Build structured content
        structured = f"""From: {msg['from']}
To: {msg['to']}
Subject: {msg['subject']}
Date: {msg['date']}

{content}"""
        
        return [ParsedDocument(
            id=self._generate_id(str(path), content),
            content=structured,
            source_type=SourceType.EMAIL,
            source_path=str(path.absolute()),
            title=msg["subject"] or "(No Subject)",
            author=msg["from"],
            created_at=self._parse_email_date(msg["date"]),
            metadata={
                "to": msg["to"],
                "cc": msg.get("cc", ""),
                "message_id": msg["message-id"],
                "in_reply_to": msg.get("in-reply-to", ""),
                "attachments": attachments,
                "thread_id": msg.get("references", "")
            }
        )]
    
    def _parse_msg(self, path: Path) -> List[ParsedDocument]:
        """Parse Outlook .msg files."""
        msg = extract_msg.Message(str(path))
        
        structured = f"""From: {msg.sender}
To: {msg.to}
Subject: {msg.subject}
Date: {msg.date}

{msg.body}"""
        
        return [ParsedDocument(
            id=self._generate_id(str(path), msg.body or ""),
            content=structured,
            source_type=SourceType.EMAIL,
            source_path=str(path.absolute()),
            title=msg.subject or "(No Subject)",
            author=msg.sender,
            created_at=msg.date,
            metadata={
                "to": msg.to,
                "cc": msg.cc or "",
                "attachments": [a.longFilename for a in msg.attachments]
            }
        )]
    
    def _parse_email_date(self, date_str) -> datetime:
        from email.utils import parsedate_to_datetime
        try:
            return parsedate_to_datetime(date_str)
        except:
            return datetime.now()

4. Web Page Parser (Knowledge Bases, Documentation Sites)

# src/parsers/web_parser.py

import httpx
import trafilatura
from urllib.parse import urlparse
from typing import List
from datetime import datetime
from .base import BaseParser, ParsedDocument, SourceType

class WebParser(BaseParser):
    """
    Parse web pages: documentation, knowledge bases, articles.
    Uses trafilatura for clean content extraction (removes nav, ads, etc).
    """
    
    def can_handle(self, source: str) -> bool:
        return source.startswith(("http://", "https://"))
    
    def parse(self, source: str) -> List[ParsedDocument]:
        # Fetch the page
        response = httpx.get(
            source, 
            follow_redirects=True,
            timeout=30,
            headers={"User-Agent": "EnterpriseRAG/1.0 (internal indexing)"}
        )
        response.raise_for_status()
        
        # Extract clean content using trafilatura
        content = trafilatura.extract(
            response.text,
            include_tables=True,
            include_links=False,
            include_comments=False,
            output_format="txt",
            favor_precision=True  # Prefer precision over recall
        )
        
        if not content:
            return []
        
        # Extract metadata
        metadata = trafilatura.extract(
            response.text,
            output_format="xmltei",  # Rich metadata format
            only_with_metadata=True
        )
        
        parsed = urlparse(source)
        
        return [ParsedDocument(
            id=self._generate_id(source, content),
            content=content,
            source_type=SourceType.WEB,
            source_path=source,
            title=self._extract_title(response.text) or parsed.path,
            created_at=datetime.now(),  # Use fetch time
            metadata={
                "domain": parsed.netloc,
                "url": source,
                "status_code": response.status_code,
                "content_type": response.headers.get("content-type", "")
            }
        )]
    
    def _extract_title(self, html: str) -> str:
        from bs4 import BeautifulSoup
        soup = BeautifulSoup(html, "html.parser")
        title_tag = soup.find("title")
        return title_tag.text.strip() if title_tag else ""

5. Database Parser (Product Catalogs, Customer Records)

# src/parsers/db_parser.py

from sqlalchemy import create_engine, text, inspect
import pandas as pd
from typing import List
from datetime import datetime
from .base import BaseParser, ParsedDocument, SourceType

class DatabaseParser(BaseParser):
    """
    Parse database records into documents.
    Each row becomes a document (or grouped by a key).
    Handles: PostgreSQL, MySQL, SQLite, SQL Server.
    """
    
    def can_handle(self, source: str) -> bool:
        # Connection strings: postgresql://, mysql://, sqlite:///
        db_prefixes = ("postgresql://", "mysql://", "sqlite:///", "mssql://")
        return source.startswith(db_prefixes)
    
    def parse(self, source: str) -> List[ParsedDocument]:
        """
        Parse all tables in a database into documents.
        Each row → one ParsedDocument.
        """
        engine = create_engine(source)
        inspector = inspect(engine)
        all_docs = []
        
        for table_name in inspector.get_table_names():
            docs = self._parse_table(engine, table_name, source)
            all_docs.extend(docs)
        
        return all_docs
    
    def _parse_table(self, engine, table_name: str, source: str) -> List[ParsedDocument]:
        """Convert each row of a table into a document."""
        df = pd.read_sql_table(table_name, engine)
        docs = []
        
        for idx, row in df.iterrows():
            # Convert row to natural language description
            content_parts = [f"Record from table '{table_name}':"]
            for col, val in row.items():
                if pd.notna(val) and str(val).strip():
                    content_parts.append(f"- {col}: {val}")
            
            content = "\n".join(content_parts)
            
            docs.append(ParsedDocument(
                id=self._generate_id(f"{source}:{table_name}:{idx}", content),
                content=content,
                source_type=SourceType.DATABASE,
                source_path=source,
                title=f"{table_name} record #{idx}",
                created_at=datetime.now(),
                metadata={
                    "table": table_name,
                    "row_index": idx,
                    "columns": list(row.index)
                }
            ))
        
        return docs

6. Slack/Teams Parser (Discussions, Decisions)

# src/parsers/slack_parser.py

from slack_sdk import WebClient
from datetime import datetime
from typing import List, Dict
from .base import BaseParser, ParsedDocument, SourceType

class SlackParser(BaseParser):
    """
    Parse Slack conversations into documents.
    Groups messages by thread for context preservation.
    Source format: "slack://CHANNEL_ID" or "slack://CHANNEL_ID/THREAD_TS"
    """
    
    def __init__(self, bot_token: str = None):
        import os
        self.client = WebClient(token=bot_token or os.getenv("SLACK_BOT_TOKEN"))
    
    def can_handle(self, source: str) -> bool:
        return source.startswith("slack://")
    
    def parse(self, source: str) -> List[ParsedDocument]:
        # Parse source: slack://CHANNEL_ID
        channel_id = source.replace("slack://", "").split("/")[0]
        
        # Fetch channel messages
        result = self.client.conversations_history(
            channel=channel_id,
            limit=1000
        )
        
        messages = result["messages"]
        
        # Group messages by thread
        threads = self._group_by_thread(channel_id, messages)
        
        # Convert each thread to a document
        docs = []
        for thread in threads:
            doc = self._thread_to_document(thread, channel_id, source)
            docs.append(doc)
        
        return docs
    
    def _group_by_thread(self, channel_id: str, messages: List[Dict]) -> List[List[Dict]]:
        """Group messages into threads (parent + replies)."""
        threads = {}
        standalone = []
        
        for msg in messages:
            thread_ts = msg.get("thread_ts")
            
            if thread_ts and thread_ts != msg.get("ts"):
                # This is a reply
                continue  # We'll fetch full threads below
            elif msg.get("reply_count", 0) > 0:
                # This is a thread parent — fetch replies
                replies = self.client.conversations_replies(
                    channel=channel_id,
                    ts=msg["ts"]
                )
                threads[msg["ts"]] = replies["messages"]
            else:
                standalone.append([msg])
        
        return list(threads.values()) + standalone
    
    def _thread_to_document(self, thread: List[Dict], channel_id: str, source: str) -> ParsedDocument:
        """Convert a thread (list of messages) to a document."""
        parts = []
        for msg in thread:
            user = msg.get("user", "unknown")
            text = msg.get("text", "")
            ts = datetime.fromtimestamp(float(msg["ts"]))
            parts.append(f"[{ts.strftime('%Y-%m-%d %H:%M')}] {user}: {text}")
        
        content = "\n".join(parts)
        first_msg = thread[0]
        
        return ParsedDocument(
            id=self._generate_id(source, content),
            content=content,
            source_type=SourceType.SLACK,
            source_path=source,
            title=first_msg.get("text", "")[:80],
            author=first_msg.get("user"),
            created_at=datetime.fromtimestamp(float(first_msg["ts"])),
            metadata={
                "channel_id": channel_id,
                "thread_ts": first_msg.get("ts"),
                "reply_count": len(thread) - 1,
                "participants": list(set(m.get("user", "") for m in thread))
            }
        )

7. Code Repository Parser (Documentation, READMEs, Code)

# src/parsers/code_parser.py

import os
from pathlib import Path
from typing import List
from datetime import datetime
from .base import BaseParser, ParsedDocument, SourceType

class CodeParser(BaseParser):
    """
    Parse code repositories: READMEs, documentation, source code.
    Extracts function signatures, docstrings, and comments.
    """
    
    CODE_EXTENSIONS = {
        ".py", ".js", ".ts", ".java", ".go", ".rs",
        ".cpp", ".c", ".rb", ".php", ".swift", ".kt"
    }
    DOC_EXTENSIONS = {".md", ".rst", ".txt", ".adoc"}
    
    def can_handle(self, source: str) -> bool:
        path = Path(source)
        return (
            path.suffix.lower() in self.CODE_EXTENSIONS | self.DOC_EXTENSIONS
            or path.name.lower() in ("readme", "readme.md", "contributing.md")
        )
    
    def parse(self, source: str) -> List[ParsedDocument]:
        path = Path(source)
        
        if path.suffix.lower() in self.DOC_EXTENSIONS:
            return self._parse_documentation(path)
        else:
            return self._parse_source_code(path)
    
    def _parse_documentation(self, path: Path) -> List[ParsedDocument]:
        """Parse markdown/text documentation files."""
        content = path.read_text(encoding="utf-8", errors="ignore")
        
        return [ParsedDocument(
            id=self._generate_id(str(path), content),
            content=content,
            source_type=SourceType.CODE,
            source_path=str(path.absolute()),
            title=path.name,
            metadata={
                "file_type": "documentation",
                "extension": path.suffix,
                "repo_path": str(path)
            }
        )]
    
    def _parse_source_code(self, path: Path) -> List[ParsedDocument]:
        """
        Parse source code: extract functions with docstrings.
        Each function/class with a docstring → one document.
        """
        content = path.read_text(encoding="utf-8", errors="ignore")
        
        if path.suffix == ".py":
            return self._parse_python_code(path, content)
        
        # For non-Python: treat entire file as one document
        return [ParsedDocument(
            id=self._generate_id(str(path), content),
            content=f"File: {path.name}\nLanguage: {path.suffix[1:]}\n\n{content}",
            source_type=SourceType.CODE,
            source_path=str(path.absolute()),
            title=path.name,
            metadata={
                "file_type": "source_code",
                "language": path.suffix[1:],
                "line_count": content.count("\n")
            }
        )]
    
    def _parse_python_code(self, path: Path, content: str) -> List[ParsedDocument]:
        """Extract Python functions/classes with docstrings."""
        import ast
        
        try:
            tree = ast.parse(content)
        except SyntaxError:
            return [ParsedDocument(
                id=self._generate_id(str(path), content),
                content=content,
                source_type=SourceType.CODE,
                source_path=str(path.absolute()),
                title=path.name,
                metadata={"file_type": "source_code", "language": "python"}
            )]
        
        docs = []
        for node in ast.walk(tree):
            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                docstring = ast.get_docstring(node)
                if docstring:
                    # Get the source code of this function/class
                    source_lines = content.split("\n")
                    func_source = "\n".join(
                        source_lines[node.lineno - 1:node.end_lineno]
                    )
                    
                    doc_content = f"""Function: {node.name}
File: {path.name}
Docstring: {docstring}

Source Code:
{func_source}"""
                    
                    docs.append(ParsedDocument(
                        id=self._generate_id(f"{path}:{node.name}", doc_content),
                        content=doc_content,
                        source_type=SourceType.CODE,
                        source_path=str(path.absolute()),
                        title=f"{path.name}::{node.name}",
                        metadata={
                            "file_type": "source_code",
                            "language": "python",
                            "symbol_type": type(node).__name__,
                            "line_number": node.lineno
                        }
                    ))
        
        # If no documented functions, return whole file
        if not docs:
            docs.append(ParsedDocument(
                id=self._generate_id(str(path), content),
                content=content,
                source_type=SourceType.CODE,
                source_path=str(path.absolute()),
                title=path.name,
                metadata={"file_type": "source_code", "language": "python"}
            ))
        
        return docs

8. OCR Parser (Scanned Documents, Papers, Handwritten)

# src/parsers/ocr_parser.py

from paddleocr import PaddleOCR
from pathlib import Path
from typing import List
from datetime import datetime
import fitz  # PyMuPDF for converting PDF pages to images
from .base import BaseParser, ParsedDocument, SourceType

class OCRParser(BaseParser):
    """
    OCR parser for scanned documents, images, and image-based PDFs.
    Uses PaddleOCR (SOTA accuracy, 80+ languages).
    Tesseract as fallback.
    """
    
    IMAGE_EXTENSIONS = {".png", ".jpg", ".jpeg", ".tiff", ".tif", ".bmp"}
    
    def __init__(self):
        self.ocr = PaddleOCR(
            use_angle_cls=True,  # Detect rotated text
            lang="en",             # Default language
            show_log=False
        )
    
    def can_handle(self, source: str) -> bool:
        path = Path(source)
        # Handle images directly or PDFs that failed text extraction
        return path.suffix.lower() in self.IMAGE_EXTENSIONS or (
            path.suffix.lower() == ".pdf" and self._is_scanned(path)
        )
    
    def parse(self, source: str) -> List[ParsedDocument]:
        path = Path(source)
        
        if path.suffix.lower() == ".pdf":
            return self._ocr_pdf(path)
        else:
            return self._ocr_image(path)
    
    def _ocr_image(self, path: Path) -> List[ParsedDocument]:
        """OCR a single image file."""
        result = self.ocr.ocr(str(path), cls=True)
        
        # Extract text with position ordering (top-to-bottom, left-to-right)
        lines = []
        confidence_scores = []
        
        for page_result in result:
            if page_result:
                # Sort by y-coordinate (top to bottom), then x (left to right)
                sorted_results = sorted(page_result, key=lambda x: (x[0][0][1], x[0][0][0]))
                for line_data in sorted_results:
                    text = line_data[1][0]
                    confidence = line_data[1][1]
                    lines.append(text)
                    confidence_scores.append(confidence)
        
        content = "\n".join(lines)
        avg_confidence = sum(confidence_scores) / len(confidence_scores) if confidence_scores else 0
        
        return [ParsedDocument(
            id=self._generate_id(str(path), content),
            content=content,
            source_type=SourceType.OCR,
            source_path=str(path.absolute()),
            title=path.stem,
            created_at=datetime.fromtimestamp(path.stat().st_mtime),
            confidence=avg_confidence,
            metadata={
                "ocr_engine": "PaddleOCR",
                "avg_confidence": avg_confidence,
                "line_count": len(lines)
            }
        )]
    
    def _ocr_pdf(self, path: Path) -> List[ParsedDocument]:
        """OCR a scanned PDF by converting pages to images first."""
        doc = fitz.open(str(path))
        all_text = []
        total_confidence = []
        
        for page_num in range(len(doc)):
            page = doc[page_num]
            # Render page as image (300 DPI for OCR quality)
            mat = fitz.Matrix(300/72, 300/72)
            pix = page.get_pixmap(matrix=mat)
            img_path = f"/tmp/ocr_page_{page_num}.png"
            pix.save(img_path)
            
            # OCR the page image
            result = self.ocr.ocr(img_path, cls=True)
            for page_result in result:
                if page_result:
                    for line_data in page_result:
                        all_text.append(line_data[1][0])
                        total_confidence.append(line_data[1][1])
            
            all_text.append(f"\n--- Page {page_num + 1} ---\n")
        
        doc.close()
        content = "\n".join(all_text)
        avg_conf = sum(total_confidence) / len(total_confidence) if total_confidence else 0
        
        return [ParsedDocument(
            id=self._generate_id(str(path), content),
            content=content,
            source_type=SourceType.OCR,
            source_path=str(path.absolute()),
            title=path.stem,
            page_count=len(doc),
            confidence=avg_conf,
            metadata={
                "ocr_engine": "PaddleOCR",
                "avg_confidence": avg_conf,
                "is_scanned_pdf": True
            }
        )]
    
    def _is_scanned(self, path: Path) -> bool:
        """Check if a PDF is image-based."""
        try:
            doc = fitz.open(str(path))
            for i in range(min(3, len(doc))):
                if len(doc[i].get_text().strip()) > 50:
                    doc.close()
                    return False
            doc.close()
            return True
        except:
            return False

9. Bill/Invoice Parser (Invoices, Receipts, Financial Documents)

# src/parsers/bill_parser.py

import re
from pathlib import Path
from typing import List, Dict
from datetime import datetime
import pdfplumber
from .base import BaseParser, ParsedDocument, SourceType

class BillParser(BaseParser):
    """
    Parse bills, invoices, and receipts.
    Extracts structured data: vendor, amount, line items, dates.
    Uses pdfplumber for precise table extraction + regex for key fields.
    """
    
    # Keywords that identify a document as a bill/invoice
    BILL_KEYWORDS = {
        "invoice", "receipt", "bill", "statement",
        "amount due", "total", "payment", "purchase order"
    }
    
    def can_handle(self, source: str) -> bool:
        path = Path(source)
        if path.suffix.lower() != ".pdf":
            return False
        
        # Check filename for bill-related keywords
        name_lower = path.stem.lower()
        return any(kw in name_lower for kw in self.BILL_KEYWORDS)
    
    def parse(self, source: str) -> List[ParsedDocument]:
        path = Path(source)
        
        with pdfplumber.open(source) as pdf:
            # Extract all text
            full_text = ""
            tables = []
            
            for page in pdf.pages:
                full_text += page.extract_text() or ""
                full_text += "\n"
                
                # Extract tables (line items)
                page_tables = page.extract_tables()
                for table in page_tables:
                    if table:
                        tables.append({
                            "headers": table[0],
                            "data": table[1:]
                        })
        
        # Extract structured fields using regex
        extracted = self._extract_bill_fields(full_text)
        
        # Create structured content
        structured_content = f"""Invoice/Bill Document
Vendor: {extracted.get('vendor', 'Unknown')}
Invoice Number: {extracted.get('invoice_number', 'N/A')}
Date: {extracted.get('date', 'N/A')}
Total Amount: {extracted.get('total', 'N/A')}
Due Date: {extracted.get('due_date', 'N/A')}

Full Content:
{full_text}"""
        
        return [ParsedDocument(
            id=self._generate_id(source, full_text),
            content=structured_content,
            source_type=SourceType.BILL,
            source_path=str(path.absolute()),
            title=f"Invoice: {extracted.get('invoice_number', path.stem)}",
            created_at=self._parse_date(extracted.get("date")),
            tables=tables,
            metadata={
                "vendor": extracted.get("vendor", ""),
                "invoice_number": extracted.get("invoice_number", ""),
                "total_amount": extracted.get("total", ""),
                "due_date": extracted.get("due_date", ""),
                "line_items_count": sum(len(t["data"]) for t in tables)
            }
        )]
    
    def _extract_bill_fields(self, text: str) -> Dict:
        """Extract key fields from invoice text using regex."""
        fields = {}
        
        # Invoice number
        inv_match = re.search(
            r"(?:invoice|inv|receipt)\s*(?:#|no\.?|number)?\s*:?\s*([A-Z0-9\-]+)",
            text, re.IGNORECASE
        )
        if inv_match:
            fields["invoice_number"] = inv_match.group(1)
        
        # Total amount
        total_match = re.search(
            r"(?:total|amount due|grand total)\s*:?\s*\$?([\d,]+\.?\d*)",
            text, re.IGNORECASE
        )
        if total_match:
            fields["total"] = f"${total_match.group(1)}"
        
        # Date
        date_match = re.search(
            r"(?:date|issued)\s*:?\s*(\d{1,2}[/\-]\d{1,2}[/\-]\d{2,4})",
            text, re.IGNORECASE
        )
        if date_match:
            fields["date"] = date_match.group(1)
        
        return fields
    
    def _parse_date(self, date_str) -> datetime:
        if not date_str:
            return datetime.now()
        for fmt in ("%m/%d/%Y", "%d/%m/%Y", "%Y-%m-%d", "%m-%d-%Y"):
            try:
                return datetime.strptime(date_str, fmt)
            except ValueError:
                continue
        return datetime.now()
