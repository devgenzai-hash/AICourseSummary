P4: Retrieval Engine

Hybrid retrieval with BM25 + vector search, RRF fusion, Cohere Rerank v3, source confidence scoring, and conversation-aware query rewriting. Sub-200ms at 10M docs.

1. Query Understanding & Rewriting

Before retrieval, we analyze the query: expand abbreviations, resolve pronouns from conversation history, and add contextual terms.

# src/retrieval/query_engine.py

from openai import OpenAI
from typing import List, Optional
from dataclasses import dataclass
from src.config import settings

@dataclass
class ConversationTurn:
    role: str       # "user" or "assistant"
    content: str

class QueryRewriter:
    """
    Rewrites user queries for optimal retrieval:
    1. Resolves pronouns from conversation history
    2. Expands abbreviations and acronyms
    3. Generates sub-queries for complex questions
    """
    
    def __init__(self):
        self.client = OpenAI(api_key=settings.openai_api_key)
    
    def rewrite(
        self,
        query: str,
        conversation: Optional[List[ConversationTurn]] = None
    ) -> List[str]:
        """
        Rewrite a query into one or more retrieval-optimized queries.
        
        Examples:
          "What about their pricing?" + history about AWS 
          → ["What is AWS pricing for EC2 instances?"]
          
          "Compare the two approaches mentioned above"
          → ["approach A details", "approach B details", "comparison A vs B"]
        """
        if not conversation:
            return [query]
        
        # Build conversation context
        history = "\n".join(
            [f"{t.role}: {t.content}" for t in conversation[-5:]]
        )
        
        response = self.client.chat.completions.create(
            model="gpt-4o-mini",  # Fast + cheap for rewriting
            messages=[
                {"role": "system", "content": """You are a search query optimizer. 
Given a conversation and a new user query:
1. Resolve all pronouns (it, they, this, that) to specific entities
2. Make the query self-contained (no context needed)
3. If the query is complex, split into 2-3 sub-queries
4. Return JSON: {"queries": ["query1", "query2"]}
Return ONLY the JSON, nothing else."""},
                {"role": "user", "content": f"Conversation:\n{history}\n\nNew query: {query}"}
            ],
            temperature=0,
            response_format={"type": "json_object"}
        )
        
        import json
        result = json.loads(response.choices[0].message.content)
        return result.get("queries", [query])

2. Hybrid Retrieval (Vector + BM25)

# src/retrieval/hybrid_search.py

from qdrant_client import QdrantClient
from qdrant_client.models import SearchParams, Filter, FieldCondition, MatchValue
from elasticsearch import Elasticsearch
from typing import List, Dict, Optional
from dataclasses import dataclass
import numpy as np
from src.config import settings
from src.ingestion.embedder import EmbeddingGenerator

@dataclass
class RetrievedChunk:
    """A retrieved chunk with scores from both retrieval paths."""
    chunk_id: str
    content: str
    source_type: str
    metadata: Dict
    vector_score: float = 0.0
    bm25_score: float = 0.0
    rrf_score: float = 0.0

class HybridSearch:
    """
    Dual-path retrieval:
    - Path A: Vector search via Qdrant (semantic similarity)
    - Path B: BM25 via Elasticsearch (keyword matching)
    - Fusion: Reciprocal Rank Fusion (RRF)
    """
    
    def __init__(self):
        self.qdrant = QdrantClient(url=settings.qdrant_url)
        self.es = Elasticsearch(settings.elasticsearch_url)
        self.embedder = EmbeddingGenerator()
        self.collection = "enterprise_rag"
        self.es_index = "enterprise_rag"
    
    def search(
        self,
        query: str,
        top_k: int = 20,
        source_filter: Optional[str] = None
    ) -> List[RetrievedChunk]:
        """
        Execute hybrid search: vector + BM25 → RRF fusion.
        
        Args:
            query: Search query
            top_k: Number of results to return
            source_filter: Optional filter by source type (e.g., "pdf", "email")
        """
        # Path A: Vector search
        vector_results = self._vector_search(query, top_k * 2, source_filter)
        
        # Path B: BM25 search
        bm25_results = self._bm25_search(query, top_k * 2, source_filter)
        
        # Fuse results with RRF
        fused = self._rrf_fusion(vector_results, bm25_results, k=60)
        
        return fused[:top_k]
    
    def _vector_search(
        self, query: str, top_k: int, source_filter: Optional[str]
    ) -> List[RetrievedChunk]:
        """Search Qdrant for semantically similar chunks."""
        query_embedding = self.embedder.embed_query(query)
        
        # Build optional filter
        qdrant_filter = None
        if source_filter:
            qdrant_filter = Filter(
                must=[FieldCondition(
                    key="source_type",
                    match=MatchValue(value=source_filter)
                )]
            )
        
        results = self.qdrant.search(
            collection_name=self.collection,
            query_vector=query_embedding.tolist(),
            limit=top_k,
            query_filter=qdrant_filter,
            search_params=SearchParams(
                hnsw_ef=128,   # Search-time accuracy (higher = slower but better)
                exact=False    # Use HNSW (not brute force)
            )
        )
        
        return [
            RetrievedChunk(
                chunk_id=hit.payload["chunk_id"],
                content=hit.payload["content"],
                source_type=hit.payload["source_type"],
                metadata=hit.payload.get("metadata", {}),
                vector_score=hit.score
            )
            for hit in results
        ]
    
    def _bm25_search(
        self, query: str, top_k: int, source_filter: Optional[str]
    ) -> List[RetrievedChunk]:
        """Search Elasticsearch for keyword-matching chunks."""
        es_query = {
            "bool": {
                "must": [{
                    "multi_match": {
                        "query": query,
                        "fields": ["content^2", "title^1.5"],
                        "type": "best_fields",
                        "fuzziness": "AUTO"
                    }
                }]
            }
        }
        
        if source_filter:
            es_query["bool"]["filter"] = [{
                "term": {"source_type": source_filter}
            }]
        
        response = self.es.search(
            index=self.es_index,
            query=es_query,
            size=top_k
        )
        
        return [
            RetrievedChunk(
                chunk_id=hit["_source"]["chunk_id"],
                content=hit["_source"]["content"],
                source_type=hit["_source"]["source_type"],
                metadata=hit["_source"].get("metadata", {}),
                bm25_score=hit["_score"]
            )
            for hit in response["hits"]["hits"]
        ]
    
    def _rrf_fusion(
        self,
        vector_results: List[RetrievedChunk],
        bm25_results: List[RetrievedChunk],
        k: int = 60
    ) -> List[RetrievedChunk]:
        """
        Reciprocal Rank Fusion (RRF).
        
        Formula: RRF(d) = Σ 1/(k + rank_i(d))
        
        k=60 is the standard constant (from the original paper).
        """
        scores = {}  # chunk_id → {rrf_score, chunk}
        
        # Score vector results by rank
        for rank, chunk in enumerate(vector_results):
            if chunk.chunk_id not in scores:
                scores[chunk.chunk_id] = {"chunk": chunk, "score": 0.0}
            scores[chunk.chunk_id]["score"] += 1.0 / (k + rank + 1)
        
        # Score BM25 results by rank
        for rank, chunk in enumerate(bm25_results):
            if chunk.chunk_id not in scores:
                scores[chunk.chunk_id] = {"chunk": chunk, "score": 0.0}
            scores[chunk.chunk_id]["score"] += 1.0 / (k + rank + 1)
        
        # Sort by RRF score
        ranked = sorted(scores.values(), key=lambda x: x["score"], reverse=True)
        
        # Assign RRF scores
        result = []
        for item in ranked:
            chunk = item["chunk"]
            chunk.rrf_score = item["score"]
            result.append(chunk)
        
        return result

3. Reranking (Cohere Rerank v3)

# src/retrieval/reranker.py

import cohere
from typing import List
from src.config import settings

class CohereReranker:
    """
    Cross-encoder reranking with Cohere Rerank v3.
    
    Why Cohere Rerank:
    - Cross-encoder architecture (much more accurate than bi-encoder)
    - Handles up to 4096 tokens per document
    - Returns calibrated relevance scores (0-1)
    - 10ms per document (fast enough for real-time)
    """
    
    def __init__(self):
        self.client = cohere.Client(settings.cohere_api_key)
        self.model = "rerank-english-v3.0"
    
    def rerank(
        self,
        query: str,
        chunks: List,
        top_k: int = 10
    ) -> List:
        """
        Rerank chunks using Cohere cross-encoder.
        
        Input: 20-40 chunks from hybrid search
        Output: Top 10 most relevant, with confidence scores
        """
        if not chunks:
            return []
        
        # Extract texts for reranking
        documents = [chunk.content for chunk in chunks]
        
        # Call Cohere Rerank
        response = self.client.rerank(
            model=self.model,
            query=query,
            documents=documents,
            top_n=top_k,
            return_documents=False
        )
        
        # Map back to our chunk objects
        reranked = []
        for result in response.results:
            chunk = chunks[result.index]
            chunk.metadata["rerank_score"] = result.relevance_score
            reranked.append(chunk)
        
        return reranked

4. Source Confidence Scoring

# src/retrieval/confidence.py

from datetime import datetime, timedelta
from typing import List, Dict
from dataclasses import dataclass

@dataclass
class ConfidenceScore:
    """Multi-dimensional confidence score for a retrieved chunk."""
    freshness: float     # 0-1: how recent is the source
    authority: float     # 0-1: how trusted is the source type
    consistency: float   # 0-1: does it agree with other chunks
    relevance: float     # 0-1: reranker score
    overall: float = 0.0 # Weighted combination
    
    def compute_overall(self):
        self.overall = (
            self.freshness * 0.15 +
            self.authority * 0.25 +
            self.consistency * 0.25 +
            self.relevance * 0.35
        )
        return self.overall

class ConfidenceScorer:
    """
    Score confidence of each retrieved chunk along 4 dimensions.
    Used to decide: which chunks to trust, which to discard,
    and whether we have enough evidence to answer.
    """
    
    # Authority weights by source type (enterprise-calibrated)
    SOURCE_AUTHORITY = {
        "pdf": 0.9,       # Official documents
        "docx": 0.85,     # Internal documents
        "database": 0.95, # Structured data = high trust
        "web": 0.6,       # External web (varies)
        "email": 0.7,     # Conversations (informal)
        "slack": 0.5,     # Very informal, may be outdated
        "code": 0.85,     # Code is factual
        "ocr": 0.65,      # OCR accuracy may vary
        "bill": 0.95,     # Financial docs = high trust
    }
    
    def score_chunks(self, chunks: List, query: str) -> List[Dict]:
        """
        Score each chunk on 4 dimensions and compute overall confidence.
        """
        scored = []
        
        for chunk in chunks:
            confidence = ConfidenceScore(
                freshness=self._score_freshness(chunk),
                authority=self._score_authority(chunk),
                consistency=self._score_consistency(chunk, chunks),
                relevance=chunk.metadata.get("rerank_score", 0.5)
            )
            confidence.compute_overall()
            
            scored.append({
                "chunk": chunk,
                "confidence": confidence
            })
        
        # Sort by overall confidence
        scored.sort(key=lambda x: x["confidence"].overall, reverse=True)
        return scored
    
    def _score_freshness(self, chunk) -> float:
        """Score based on document age. Newer = higher."""
        created = chunk.metadata.get("created_at")
        if not created:
            return 0.5  # Unknown age
        
        if isinstance(created, str):
            created = datetime.fromisoformat(created)
        
        age_days = (datetime.now() - created).days
        
        if age_days < 7:     return 1.0
        elif age_days < 30:  return 0.9
        elif age_days < 90:  return 0.75
        elif age_days < 365: return 0.5
        else:               return 0.3
    
    def _score_authority(self, chunk) -> float:
        """Score based on source type trustworthiness."""
        return self.SOURCE_AUTHORITY.get(chunk.source_type, 0.5)
    
    def _score_consistency(self, chunk, all_chunks: List) -> float:
        """
        Score based on agreement with other retrieved chunks.
        If multiple sources say the same thing → high consistency.
        """
        if len(all_chunks) < 2:
            return 0.5
        
        # Simple heuristic: count overlapping key terms
        chunk_terms = set(chunk.content.lower().split())
        overlap_scores = []
        
        for other in all_chunks:
            if other.chunk_id == chunk.chunk_id:
                continue
            other_terms = set(other.content.lower().split())
            overlap = len(chunk_terms & other_terms) / max(len(chunk_terms), 1)
            overlap_scores.append(overlap)
        
        return min(sum(overlap_scores) / max(len(overlap_scores), 1), 1.0)

5. Complete Retrieval Pipeline

# src/retrieval/retriever.py

from typing import List, Optional, Dict
from dataclasses import dataclass
from .query_engine import QueryRewriter, ConversationTurn
from .hybrid_search import HybridSearch, RetrievedChunk
from .reranker import CohereReranker
from .confidence import ConfidenceScorer

@dataclass
class RetrievalResult:
    """Final retrieval output sent to the generation stage."""
    chunks: List[Dict]           # Chunks with confidence scores
    queries_used: List[str]     # Rewritten queries
    avg_confidence: float       # Mean confidence across top chunks
    has_sufficient_evidence: bool  # Can we answer reliably?

class HybridRetriever:
    """
    Full retrieval pipeline:
    
    Query → Rewrite → Hybrid Search → Rerank → Confidence Score → Filter
    
    Returns scored chunks ready for generation.
    """
    
    def __init__(self):
        self.rewriter = QueryRewriter()
        self.searcher = HybridSearch()
        self.reranker = CohereReranker()
        self.scorer = ConfidenceScorer()
        
        # Thresholds
        self.min_confidence = 0.4     # Drop chunks below this
        self.evidence_threshold = 0.6 # Min avg confidence to answer
    
    def retrieve(
        self,
        query: str,
        conversation: Optional[List[ConversationTurn]] = None,
        source_filter: Optional[str] = None,
        top_k: int = 10
    ) -> RetrievalResult:
        """
        Execute the full retrieval pipeline.
        
        Args:
            query: User's question
            conversation: Previous turns (for pronoun resolution)
            source_filter: Optional source type filter
            top_k: Number of final chunks to return
        
        Returns:
            RetrievalResult with scored chunks and confidence metrics
        """
        # Step 1: Rewrite query (resolve pronouns, expand)
        rewritten_queries = self.rewriter.rewrite(query, conversation)
        
        # Step 2: Hybrid search for each sub-query
        all_chunks = []
        for q in rewritten_queries:
            results = self.searcher.search(q, top_k=20, source_filter=source_filter)
            all_chunks.extend(results)
        
        # Deduplicate (same chunk may appear for multiple sub-queries)
        seen_ids = set()
        unique_chunks = []
        for chunk in all_chunks:
            if chunk.chunk_id not in seen_ids:
                seen_ids.add(chunk.chunk_id)
                unique_chunks.append(chunk)
        
        # Step 3: Rerank top candidates
        reranked = self.reranker.rerank(query, unique_chunks[:30], top_k=top_k)
        
        # Step 4: Score confidence
        scored = self.scorer.score_chunks(reranked, query)
        
        # Step 5: Filter low-confidence chunks
        filtered = [
            s for s in scored
            if s["confidence"].overall >= self.min_confidence
        ]
        
        # Compute overall evidence quality
        avg_conf = (
            sum(s["confidence"].overall for s in filtered) / len(filtered)
            if filtered else 0.0
        )
        
        return RetrievalResult(
            chunks=filtered,
            queries_used=rewritten_queries,
            avg_confidence=avg_conf,
            has_sufficient_evidence=avg_conf >= self.evidence_threshold and len(filtered) >= 2
        )

Performance Characteristics

Stage Latency Notes
Query rewriting (GPT-4o-mini) ~80ms Only needed with conversation context
Query embedding (BGE-M3) ~15ms Single query encoding
Qdrant vector search ~20ms HNSW with INT8 quantization
Elasticsearch BM25 ~15ms 3-shard index, cached filters
RRF fusion <1ms In-memory rank combination
Cohere Rerank (20 docs) ~60ms Cross-encoder, API call
Confidence scoring <5ms In-memory computation
Total (cold) ~195ms All stages, no cache
Total (warm, no rewrite) ~115ms Skip rewriting, warmed connections
✅ Key Design Decisions
  • RRF over linear combination — rank-based fusion is more robust than score normalization
  • Cohere over self-hosted cross-encoder — 10x lower latency vs running on CPU, consistent quality
  • Confidence scoring BEFORE generation — we can refuse to answer BEFORE wasting an LLM call
  • Multi-query for complex questions — decomposes "Compare X and Y" into parallel retrievals
