1. Deduplication Engine

At 10M documents, you will have duplicates. Same document shared via email, Slack, and saved in two folders. We need both exact and near-duplicate detection.

# src/ingestion/deduplicator.py

import hashlib
from datasketch import MinHash, MinHashLSH
from typing import List, Set, Tuple
from src.parsers.base import ParsedDocument

class Deduplicator:
    """
    Two-layer deduplication:
    1. Exact: SHA256 hash of content (catches identical copies)
    2. Near-duplicate: MinHash LSH (catches slightly modified copies)
    """
    
    def __init__(self, similarity_threshold: float = 0.85):
        self.seen_hashes: Set[str] = set()
        self.lsh = MinHashLSH(threshold=similarity_threshold, num_perm=128)
        self.minhash_store = {}  # doc_id → MinHash
    
    def is_duplicate(self, doc: ParsedDocument) -> Tuple[bool, str]:
        """
        Check if document is a duplicate.
        Returns (is_dup, reason) — reason is 'exact', 'near', or ''.
        """
        # Layer 1: Exact hash match
        content_hash = hashlib.sha256(doc.content.encode()).hexdigest()
        if content_hash in self.seen_hashes:
            return True, "exact"
        
        # Layer 2: Near-duplicate via MinHash LSH
        minhash = self._compute_minhash(doc.content)
        
        try:
            similar_docs = self.lsh.query(minhash)
            if similar_docs:
                return True, "near"
        except ValueError:
            pass
        
        # Not a duplicate — register it
        self.seen_hashes.add(content_hash)
        self.lsh.insert(doc.id, minhash)
        self.minhash_store[doc.id] = minhash
        
        return False, ""
    
    def deduplicate_batch(self, docs: List[ParsedDocument]) -> List[ParsedDocument]:
        """Remove duplicates from a batch of documents."""
        unique = []
        stats = {"exact": 0, "near": 0, "unique": 0}
        
        for doc in docs:
            is_dup, reason = self.is_duplicate(doc)
            if not is_dup:
                unique.append(doc)
                stats["unique"] += 1
            else:
                stats[reason] += 1
        
        print(f"📊 Dedup: {stats['unique']} unique, {stats['exact']} exact dups, {stats['near']} near dups")
        return unique
    
    def _compute_minhash(self, text: str) -> MinHash:
        """Compute MinHash signature from text shingles."""
        m = MinHash(num_perm=128)
        # Use 3-word shingles
        words = text.lower().split()
        for i in range(len(words) - 2):
            shingle = " ".join(words[i:i+3])
            m.update(shingle.encode("utf-8"))
        return m

2. Text Normalization

# src/ingestion/normalizer.py

import re
import unicodedata

class TextNormalizer:
    """
    Clean and normalize text before chunking.
    Handles encoding issues, whitespace, special characters.
    """
    
    def normalize(self, text: str) -> str:
        # 1. Unicode normalization (NFC form)
        text = unicodedata.normalize("NFC", text)
        
        # 2. Replace smart quotes with standard quotes
        text = text.replace("\u2018", "'").replace("\u2019", "'")
        text = text.replace("\u201c", '"').replace("\u201d", '"')
        
        # 3. Replace em/en dashes
        text = text.replace("\u2013", "-").replace("\u2014", " - ")
        
        # 4. Collapse multiple whitespace (preserve single newlines)
        text = re.sub(r"[ \t]+", " ", text)
        text = re.sub(r"\n{3,}", "\n\n", text)
        
        # 5. Remove null bytes and control characters
        text = re.sub(r"[\x00-\x08\x0b\x0c\x0e-\x1f]", "", text)
        
        # 6. Strip leading/trailing whitespace per line
        lines = [line.strip() for line in text.split("\n")]
        text = "\n".join(lines)
        
        return text.strip()

3. Smart Chunking (Per Document Type)

⚠️ Critical Insight

Chunking strategy should differ by document type. A PDF research paper needs section-aware chunking. An email thread needs per-message chunks. Code needs function-level chunks. One-size-fits-all chunking destroys context.

# src/ingestion/chunker.py

from dataclasses import dataclass
from typing import List
from src.parsers.base import ParsedDocument, SourceType

@dataclass
class Chunk:
    """A chunk of text ready for embedding."""
    id: str
    content: str
    doc_id: str            # Parent document ID
    chunk_index: int
    source_type: SourceType
    metadata: dict         # Inherited from parent + chunk-specific

class SmartChunker:
    """
    Chunk documents using type-specific strategies.
    
    - PDF/Papers: Section-aware (split at headings, 512-token chunks)
    - Emails: Per-message (each email = one chunk)
    - Slack: Per-thread (conversation context preserved)
    - Code: Per-function/class
    - Bills: Whole document (structured, usually short)
    - Default: Sliding window with overlap
    """
    
    def __init__(self, max_tokens: int = 512, overlap_tokens: int = 50):
        self.max_tokens = max_tokens
        self.overlap = overlap_tokens
    
    def chunk(self, doc: ParsedDocument) -> List[Chunk]:
        """Route to type-specific chunking strategy."""
        strategy_map = {
            SourceType.PDF: self._chunk_by_sections,
            SourceType.DOCX: self._chunk_by_sections,
            SourceType.EMAIL: self._chunk_email,
            SourceType.SLACK: self._chunk_conversation,
            SourceType.CODE: self._chunk_code,
            SourceType.BILL: self._chunk_whole,
            SourceType.DATABASE: self._chunk_whole,
            SourceType.WEB: self._chunk_by_sections,
            SourceType.OCR: self._chunk_sliding_window,
        }
        
        strategy = strategy_map.get(doc.source_type, self._chunk_sliding_window)
        chunks = strategy(doc)
        
        # Ensure no chunk exceeds max size
        final_chunks = []
        for chunk in chunks:
            if self._token_count(chunk.content) > self.max_tokens * 1.5:
                # Split oversized chunks
                sub_chunks = self._split_large_chunk(chunk)
                final_chunks.extend(sub_chunks)
            else:
                final_chunks.append(chunk)
        
        return final_chunks
    
    def _chunk_by_sections(self, doc: ParsedDocument) -> List[Chunk]:
        """Split by document sections (headings), then sliding window within."""
        chunks = []
        
        if doc.sections:
            for i, section in enumerate(doc.sections):
                section_text = f"{section['heading']}\n\n{section['content']}"
                
                if self._token_count(section_text) <= self.max_tokens:
                    chunks.append(Chunk(
                        id=f"{doc.id}_s{i}",
                        content=section_text,
                        doc_id=doc.id,
                        chunk_index=i,
                        source_type=doc.source_type,
                        metadata={
                            "section": section["heading"],
                            "title": doc.title,
                            "source": doc.source_path
                        }
                    ))
                else:
                    # Section too large — apply sliding window
                    sub_chunks = self._sliding_window(
                        section_text, doc, start_index=len(chunks)
                    )
                    chunks.extend(sub_chunks)
        else:
            # No sections — fallback to sliding window
            chunks = self._sliding_window(doc.content, doc, start_index=0)
        
        return chunks
    
    def _chunk_email(self, doc: ParsedDocument) -> List[Chunk]:
        """Each email message is one chunk (preserves full context)."""
        return [Chunk(
            id=f"{doc.id}_e0",
            content=doc.content,
            doc_id=doc.id,
            chunk_index=0,
            source_type=doc.source_type,
            metadata={
                "title": doc.title,
                "author": doc.author,
                "source": doc.source_path,
                **doc.metadata
            }
        )]
    
    def _chunk_conversation(self, doc: ParsedDocument) -> List[Chunk]:
        """Slack threads: use a sliding window of messages."""
        messages = doc.content.split("\n")
        chunks = []
        window_size = 10  # 10 messages per chunk
        step = 7           # 3-message overlap
        
        for i in range(0, len(messages), step):
            window = messages[i:i + window_size]
            content = "\n".join(window)
            
            chunks.append(Chunk(
                id=f"{doc.id}_c{len(chunks)}",
                content=content,
                doc_id=doc.id,
                chunk_index=len(chunks),
                source_type=doc.source_type,
                metadata={
                    "title": doc.title,
                    "source": doc.source_path,
                    "message_range": f"{i}-{i+len(window)}"
                }
            ))
        
        return chunks
    
    def _chunk_code(self, doc: ParsedDocument) -> List[Chunk]:
        """Code files: already split by function in parsing. Keep as-is."""
        return [Chunk(
            id=f"{doc.id}_code0",
            content=doc.content,
            doc_id=doc.id,
            chunk_index=0,
            source_type=doc.source_type,
            metadata={"title": doc.title, "source": doc.source_path}
        )]
    
    def _chunk_whole(self, doc: ParsedDocument) -> List[Chunk]:
        """Bills/DB records: keep as single chunk (usually short)."""
        return [Chunk(
            id=f"{doc.id}_w0",
            content=doc.content,
            doc_id=doc.id,
            chunk_index=0,
            source_type=doc.source_type,
            metadata={"title": doc.title, "source": doc.source_path, **doc.metadata}
        )]
    
    def _chunk_sliding_window(self, doc: ParsedDocument) -> List[Chunk]:
        """Default: sliding window chunking."""
        return self._sliding_window(doc.content, doc, 0)
    
    def _sliding_window(self, text: str, doc: ParsedDocument, start_index: int) -> List[Chunk]:
        """Apply sliding window with overlap."""
        words = text.split()
        chunks = []
        step = self.max_tokens - self.overlap
        
        for i in range(0, len(words), step):
            chunk_words = words[i:i + self.max_tokens]
            content = " ".join(chunk_words)
            
            chunks.append(Chunk(
                id=f"{doc.id}_sw{start_index + len(chunks)}",
                content=content,
                doc_id=doc.id,
                chunk_index=start_index + len(chunks),
                source_type=doc.source_type,
                metadata={"title": doc.title, "source": doc.source_path}
            ))
        
        return chunks
    
    def _token_count(self, text: str) -> int:
        """Approximate token count (words ≈ 0.75 tokens)."""
        return int(len(text.split()) * 1.3)

4. Embedding Generator (BGE-M3)

# src/ingestion/embedder.py

import numpy as np
from sentence_transformers import SentenceTransformer
from typing import List
from src.config import settings

class EmbeddingGenerator:
    """
    Generate embeddings using BAAI/bge-m3.
    
    Why BGE-M3:
    - 1024 dimensions (good balance of quality vs storage)
    - Multilingual (100+ languages)
    - Open-source (no API cost for embeddings)
    - SOTA on MTEB retrieval benchmarks
    """
    
    def __init__(self):
        self.model = SentenceTransformer(
            settings.embedding_model,
            device="cuda"  # Use GPU if available
        )
        self.dimension = settings.embedding_dimension
        self.batch_size = settings.embedding_batch_size
    
    def embed_chunks(self, chunks: List) -> List[np.ndarray]:
        """
        Embed a list of chunks in batches.
        
        At 10M docs with ~5 chunks each = 50M embeddings.
        At 1024 dims × 4 bytes = ~200GB of vectors.
        Batch processing is essential.
        """
        texts = [chunk.content for chunk in chunks]
        
        all_embeddings = []
        for i in range(0, len(texts), self.batch_size):
            batch = texts[i:i + self.batch_size]
            embeddings = self.model.encode(
                batch,
                normalize_embeddings=True,  # L2 normalize for cosine sim
                show_progress_bar=False,
                batch_size=self.batch_size
            )
            all_embeddings.extend(embeddings)
            
            if i % (1000) == 0 and i > 0:
                print(f"  Embedded {i}/{len(texts)} chunks...")
        
        return all_embeddings
    
    def embed_query(self, query: str) -> np.ndarray:
        """Embed a single query (adds 'query: ' prefix for BGE models)."""
        return self.model.encode(
            f"query: {query}",
            normalize_embeddings=True
        )

5. Dual Indexer (Qdrant + Elasticsearch)

# src/ingestion/indexer.py

from qdrant_client import QdrantClient
from qdrant_client.models import (
    Distance, VectorParams, PointStruct,
    OptimizersConfigDiff, HnswConfigDiff, 
    ScalarQuantization, ScalarQuantizationConfig, ScalarType
)
from elasticsearch import Elasticsearch, helpers
from typing import List
import numpy as np
from src.config import settings

class DualIndexer:
    """
    Index chunks into both:
    - Qdrant (vector similarity search)
    - Elasticsearch (BM25 keyword search)
    
    This enables hybrid retrieval (Stage 2).
    """
    
    COLLECTION_NAME = "enterprise_rag"
    ES_INDEX_NAME = "enterprise_rag"
    
    def __init__(self):
        # Qdrant connection
        self.qdrant = QdrantClient(url=settings.qdrant_url)
        
        # Elasticsearch connection
        self.es = Elasticsearch(settings.elasticsearch_url)
        
        # Initialize collections/indexes
        self._init_qdrant()
        self._init_elasticsearch()
    
    def _init_qdrant(self):
        """Create Qdrant collection optimized for 10M vectors."""
        collections = self.qdrant.get_collections().collections
        exists = any(c.name == self.COLLECTION_NAME for c in collections)
        
        if not exists:
            self.qdrant.create_collection(
                collection_name=self.COLLECTION_NAME,
                vectors_config=VectorParams(
                    size=settings.embedding_dimension,  # 1024 for BGE-M3
                    distance=Distance.COSINE,
                ),
                # HNSW config optimized for 10M+ vectors
                hnsw_config=HnswConfigDiff(
                    m=16,                  # Connections per node
                    ef_construct=200,      # Build-time accuracy
                    full_scan_threshold=20000,
                ),
                # Scalar quantization: reduces memory 4x
                quantization_config=ScalarQuantization(
                    scalar=ScalarQuantizationConfig(
                        type=ScalarType.INT8,
                        quantile=0.99,
                        always_ram=True
                    )
                ),
                optimizers_config=OptimizersConfigDiff(
                    indexing_threshold=50000,  # Index after 50K points
                )
            )
            print(f"✅ Created Qdrant collection: {self.COLLECTION_NAME}")
    
    def _init_elasticsearch(self):
        """Create Elasticsearch index with BM25 mapping."""
        if not self.es.indices.exists(index=self.ES_INDEX_NAME):
            mapping = {
                "settings": {
                    "number_of_shards": 3,     # For 10M docs
                    "number_of_replicas": 1,
                    "analysis": {
                        "analyzer": {
                            "rag_analyzer": {
                                "type": "custom",
                                "tokenizer": "standard",
                                "filter": ["lowercase", "stop", "snowball"]
                            }
                        }
                    }
                },
                "mappings": {
                    "properties": {
                        "chunk_id": {"type": "keyword"},
                        "doc_id": {"type": "keyword"},
                        "content": {
                            "type": "text",
                            "analyzer": "rag_analyzer"
                        },
                        "source_type": {"type": "keyword"},
                        "title": {"type": "text"},
                        "source_path": {"type": "keyword"},
                        "created_at": {"type": "date"},
                        "metadata": {"type": "object", "enabled": False}
                    }
                }
            }
            self.es.indices.create(index=self.ES_INDEX_NAME, body=mapping)
            print(f"✅ Created Elasticsearch index: {self.ES_INDEX_NAME}")
    
    def index_batch(self, chunks: List, embeddings: List[np.ndarray]):
        """Index a batch of chunks into both stores."""
        # === Qdrant: Vector indexing ===
        points = [
            PointStruct(
                id=hash(chunk.id) % (2**63),  # Qdrant needs int IDs
                vector=embedding.tolist(),
                payload={
                    "chunk_id": chunk.id,
                    "doc_id": chunk.doc_id,
                    "content": chunk.content,
                    "source_type": chunk.source_type.value,
                    "metadata": chunk.metadata
                }
            )
            for chunk, embedding in zip(chunks, embeddings)
        ]
        
        # Upload in batches of 100
        for i in range(0, len(points), 100):
            self.qdrant.upsert(
                collection_name=self.COLLECTION_NAME,
                points=points[i:i+100]
            )
        
        # === Elasticsearch: Text indexing ===
        actions = [
            {
                "_index": self.ES_INDEX_NAME,
                "_id": chunk.id,
                "_source": {
                    "chunk_id": chunk.id,
                    "doc_id": chunk.doc_id,
                    "content": chunk.content,
                    "source_type": chunk.source_type.value,
                    "title": chunk.metadata.get("title", ""),
                    "source_path": chunk.metadata.get("source", ""),
                    "metadata": chunk.metadata
                }
            }
            for chunk in chunks
        ]
        
        helpers.bulk(self.es, actions)
        print(f"  Indexed {len(chunks)} chunks → Qdrant + Elasticsearch")

6. Full Ingestion Pipeline Orchestrator

# src/ingestion/pipeline.py

from typing import List
from src.parsers.router import DocumentRouter
from .deduplicator import Deduplicator
from .normalizer import TextNormalizer
from .chunker import SmartChunker
from .embedder import EmbeddingGenerator
from .indexer import DualIndexer

class IngestionPipeline:
    """
    Full ingestion orchestrator.
    
    Flow: Parse → Dedup → Normalize → Chunk → Embed → Index
    
    Handles 10M documents via batch processing.
    """
    
    def __init__(self):
        self.router = DocumentRouter()
        self.dedup = Deduplicator()
        self.normalizer = TextNormalizer()
        self.chunker = SmartChunker()
        self.embedder = EmbeddingGenerator()
        self.indexer = DualIndexer()
    
    def ingest_directory(self, directory: str) -> dict:
        """Ingest all documents from a directory."""
        print(f"\n🚀 Starting ingestion from: {directory}")
        
        # Step 1: Parse all documents
        print("📄 Step 1: Parsing documents...")
        docs = self.router.parse_directory(directory)
        
        # Step 2: Deduplicate
        print("🔍 Step 2: Deduplicating...")
        unique_docs = self.dedup.deduplicate_batch(docs)
        
        # Step 3: Normalize text
        print("✨ Step 3: Normalizing...")
        for doc in unique_docs:
            doc.content = self.normalizer.normalize(doc.content)
        
        # Step 4: Chunk documents
        print("✂️ Step 4: Chunking...")
        all_chunks = []
        for doc in unique_docs:
            chunks = self.chunker.chunk(doc)
            all_chunks.extend(chunks)
        print(f"   Created {len(all_chunks)} chunks from {len(unique_docs)} docs")
        
        # Step 5: Generate embeddings (batch)
        print("🧠 Step 5: Generating embeddings...")
        embeddings = self.embedder.embed_chunks(all_chunks)
        
        # Step 6: Index into Qdrant + Elasticsearch
        print("💾 Step 6: Indexing...")
        batch_size = 500
        for i in range(0, len(all_chunks), batch_size):
            batch_chunks = all_chunks[i:i + batch_size]
            batch_embeddings = embeddings[i:i + batch_size]
            self.indexer.index_batch(batch_chunks, batch_embeddings)
        
        stats = {
            "documents_parsed": len(docs),
            "documents_unique": len(unique_docs),
            "duplicates_removed": len(docs) - len(unique_docs),
            "chunks_created": len(all_chunks),
            "embeddings_generated": len(embeddings)
        }
        
        print(f"\n✅ Ingestion complete! {stats}")
        return stats

# === CLI Entry Point ===
if __name__ == "__main__":
    import sys
    directory = sys.argv[1] if len(sys.argv) > 1 else "./data/documents"
    pipeline = IngestionPipeline()
    pipeline.ingest_directory(directory)
📊 10M Document Ingestion Math
  • Documents: 10,000,000
  • Avg chunks per doc: 5 → 50,000,000 chunks
  • Embedding time: ~50M × 5ms = ~70 hours on single GPU (parallelize!)
  • Vector storage: 50M × 1024 dims × 4 bytes = ~200GB (with INT8 quant: ~50GB)
  • Elasticsearch: ~50M documents × ~500 bytes avg = ~25GB
  • Strategy: Process in parallel batches of 10K docs. Use multi-GPU for embeddings.
