1. What is Image Classification?

Image classification is the task of assigning a label (class) to an input image from a fixed set of categories. For example, given images of cats and dogs, the model learns to predict whether a new image is a "cat" or a "dog".

The Core Idea

You show the model many labeled examples → It learns patterns → It predicts labels for new unseen images.

Input: Raw pixel values (e.g., 224×224×3 = 150,528 numbers for a color image)

Output: Probability distribution over classes (e.g., [0.95 cat, 0.05 dog])

import os
import shutil
import random
from pathlib import Path

def organize_dataset(source_dir, output_dir, train_ratio=0.7, val_ratio=0.15):
    """
    Organize images into train/val/test splits.
    
    Args:
        source_dir: Folder with subfolders for each class
                    e.g., source_dir/cats/*.jpg, source_dir/dogs/*.jpg
        output_dir: Where to create train/val/test structure
        train_ratio: Fraction for training (default 70%)
        val_ratio: Fraction for validation (default 15%)
    """
    random.seed(42)  # Reproducibility
    
    classes = [d for d in os.listdir(source_dir) 
               if os.path.isdir(os.path.join(source_dir, d))]
    
    for split in ['train', 'val', 'test']:
        for cls in classes:
            os.makedirs(os.path.join(output_dir, split, cls), exist_ok=True)
    
    for cls in classes:
        cls_path = os.path.join(source_dir, cls)
        images = [f for f in os.listdir(cls_path) 
                  if f.lower().endswith(('.jpg', '.jpeg', '.png', '.bmp', '.webp'))]
        
        random.shuffle(images)
        
        n_train = int(len(images) * train_ratio)
        n_val = int(len(images) * val_ratio)
        
        splits = {
            'train': images[:n_train],
            'val': images[n_train:n_train + n_val],
            'test': images[n_train + n_val:]
        }
        
        for split_name, split_images in splits.items():
            for img in split_images:
                src = os.path.join(cls_path, img)
                dst = os.path.join(output_dir, split_name, cls, img)
                shutil.copy2(src, dst)
        
        print(f"  {cls}: {len(splits['train'])} train, "
              f"{len(splits['val'])} val, {len(splits['test'])} test")

4. Image Preprocessing & Augmentation

Why Preprocess?

  • Resize: Models need fixed input dimensions (e.g., 224×224)
  • Normalize: Scale pixel values from [0, 255] to [0, 1] or [-1, 1] for faster convergence
  • Augment: Artificially increase dataset size by applying transformations

🔑 Key Concept: Data Augmentation

Data augmentation creates modified copies of your images during training. This helps the model:

  • Become invariant to position, rotation, lighting
  • Reduce overfitting (model memorizing training data)
  • Effectively increase dataset size 5-10x

Common Augmentation Techniques

5. Feature Extraction (Traditional ML)

Traditional ML algorithms (SVM, KNN, Random Forest) cannot directly process raw pixel arrays efficiently. We need to extract meaningful features first.

🧠 What are Features?

Features are numerical representations that capture meaningful information from images:

  • Color Histograms: Distribution of colors in the image
  • HOG (Histogram of Oriented Gradients): Captures edge/shape information
  • LBP (Local Binary Patterns): Captures texture patterns
  • SIFT/SURF: Detects distinctive keypoints
  • Flattened Pixels: Simply reshape image into 1D array (least effective)

Code: Feature Extraction Methods

import cv2
import numpy as np
from skimage.feature import hog, local_binary_pattern
from sklearn.preprocessing import StandardScaler
import os

def extract_color_histogram(image, bins=32):
    """
    Extract color histogram features.
    Captures the distribution of colors - good for scenes/objects with distinct colors.
    """
    features = []
    for channel in range(3):  # B, G, R channels
        hist = cv2.calcHist([image], [channel], None, [bins], [0, 256])
        hist = hist.flatten()
        features.extend(hist)
    return np.array(features)  # Shape: (96,) for bins=32

def extract_hog_features(image, orientations=9, pixels_per_cell=(8,8), cells_per_block=(2,2)):
    """
    Extract HOG (Histogram of Oriented Gradients) features.
    Captures shape/edge information - excellent for object recognition.
    
    How it works:
    1. Compute gradients (edge directions) in small cells
    2. Build histogram of gradient orientations per cell
    3. Normalize across blocks for illumination invariance
    """
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    gray = cv2.resize(gray, (128, 128))
    
    features = hog(
        gray,
        orientations=orientations,
        pixels_per_cell=pixels_per_cell,
        cells_per_block=cells_per_block,
        block_norm='L2-Hys',
        visualize=False,
        feature_vector=True
    )
    return features  # Shape: (8100,) for default params with 128x128

def extract_lbp_features(image, radius=3, n_points=24, bins=256):
    """
    Extract LBP (Local Binary Pattern) features.
    Captures texture information - good for materials, textures.
    
    How it works:
    1. For each pixel, compare with surrounding neighbors
    2. Create binary pattern based on greater/less than center
    3. Build histogram of these patterns
    """
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    gray = cv2.resize(gray, (128, 128))
    
    lbp = local_binary_pattern(gray, n_points, radius, method='uniform')
    hist, _ = np.histogram(lbp.ravel(), bins=bins, range=(0, bins))
    hist = hist.astype(np.float64)
    hist /= (hist.sum() + 1e-7)  # Normalize
    return hist

def extract_combined_features(image):
    """Combine multiple feature types for better representation."""
    color_feat = extract_color_histogram(image)
    hog_feat = extract_hog_features(image)
    lbp_feat = extract_lbp_features(image)
    
    # Concatenate all features
    combined = np.concatenate([color_feat, hog_feat, lbp_feat])
    return combined

def load_dataset_with_features(data_dir, feature_fn=extract_combined_features, img_size=(128, 128)):
    """
    Load all images and extract features.
    Returns feature matrix X and label vector y.
    """
    X = []
    y = []
    class_names = sorted(os.listdir(data_dir))
    
    for label_idx, class_name in enumerate(class_names):
        class_dir = os.path.join(data_dir, class_name)
        if not os.path.isdir(class_dir):
            continue
            
        for img_file in os.listdir(class_dir):
            img_path = os.path.join(class_dir, img_file)
            image = cv2.imread(img_path)
            if image is None:
                continue
            
            image = cv2.resize(image, img_size)
            features = feature_fn(image)
            X.append(features)
            y.append(label_idx)
    
    X = np.array(X)
    y = np.array(y)
    
    # Standardize features (zero mean, unit variance)
    scaler = StandardScaler()
    X = scaler.fit_transform(X)
    
    print(f"Dataset: {X.shape[0]} images, {X.shape[1]} features, {len(class_names)} classes")
    print(f"Classes: {class_names}")
    return X, y, class_names, scaler
📊 Method 1: Traditional ML (SVM, KNN, Random Forest)

When to Use

  • Small dataset (50-500 images per class)
  • Limited computing resources (no GPU needed)
  • Need fast training and interpretable results
  • Images have distinctive color/texture/shape patterns

How These Algorithms Work

SVM (Support Vector Machine): Finds the optimal hyperplane that separates classes with maximum margin. Uses kernel tricks to handle non-linear boundaries. Best overall for traditional image classification.

KNN (K-Nearest Neighbors): For a new image, finds K most similar images in training set and votes. Simple but slow at prediction time with large datasets.

Random Forest: Ensemble of decision trees, each trained on random subsets. Robust to noise, handles high-dimensional features well.

🧠 Method 2: CNN from Scratch (Convolutional Neural Network)

How CNNs Work

CNN Architecture Explained

A CNN automatically learns hierarchical features from images:

  • Convolutional Layers: Slide small filters over the image to detect patterns (edges → textures → parts → objects)
  • Activation (ReLU): Introduces non-linearity: f(x) = max(0, x)
  • Pooling Layers: Downsample feature maps (reduce size, keep important info)
  • Flatten: Convert 2D feature maps to 1D vector
  • Dense Layers: Fully connected layers for final classification
  • Softmax: Convert final output to probability distribution
Input Image (224×224×3) │ ▼ ┌─────────────────────────────────────────────────────────────────┐ │ CONV Layer 1 (32 filters, 3×3) → ReLU → MaxPool (2×2) │ │ Output: 112×112×32 │ ├─────────────────────────────────────────────────────────────────┤ │ CONV Layer 2 (64 filters, 3×3) → ReLU → MaxPool (2×2) │ │ Output: 56×56×64 │ ├─────────────────────────────────────────────────────────────────┤ │ CONV Layer 3 (128 filters, 3×3) → ReLU → MaxPool (2×2) │ │ Output: 28×28×128 │ ├─────────────────────────────────────────────────────────────────┤ │ CONV Layer 4 (256 filters, 3×3) → ReLU → MaxPool (2×2) │ │ Output: 14×14×256 │ ├─────────────────────────────────────────────────────────────────┤ │ Flatten → 14×14×256 = 50,176 neurons │ ├─────────────────────────────────────────────────────────────────┤ │ Dense (512) → ReLU → Dropout(0.5) → Dense(num_classes) │ │ → Softmax │ └─────────────────────────────────────────────────────────────────┘ │ ▼ Output: [0.02, 0.95, 0.03] → "Cat" (95% confidence)

Key Concepts

🔍 Convolution Operation

A small matrix (kernel/filter) slides over the input image. At each position, element-wise multiplication + sum produces one output value. Different filters detect different patterns:

  • Early layers: edges, corners, colors
  • Middle layers: textures, patterns, parts
  • Deep layers: full objects, scenes

📉 Loss Function & Backpropagation

Categorical Cross-Entropy Loss: Measures how far predictions are from true labels.

Loss = -Σ y_true × log(y_predicted)

Backpropagation: Computes gradients (how much each weight contributed to the error) using chain rule, then updates weights to reduce loss.

Optimizer (Adam): Adaptive learning rate optimizer. Adjusts step size per parameter based on past gradients.

def build_cnn(input_shape=(224, 224, 3), num_classes=3):
    model = models.Sequential([
        # Block 1
        layers.Conv2D(32, (3, 3), padding='same', input_shape=input_shape),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.Conv2D(32, (3, 3), padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.MaxPooling2D((2, 2)),
        layers.Dropout(0.25),
        
        # Block 2
        layers.Conv2D(64, (3, 3), padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.Conv2D(64, (3, 3), padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.MaxPooling2D((2, 2)),
        layers.Dropout(0.25),
        
        # Block 3
        layers.Conv2D(128, (3, 3), padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.Conv2D(128, (3, 3), padding='same'),
        layers.BatchNormalization(),

        layers.BatchNormalization(),
        layers.Activation('relu'),

    ])

def train_one_epoch(model, loader, criterion, optimizer, device):
    model.train()
    running_loss = 0.0
    correct = 0
    total = 0
    
    for images, labels in loader:
        images, labels = images.to(device), labels.to(device)
        
        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
        _, predicted = outputs.max(1)
        total += labels.size(0)
        correct += predicted.eq(labels).sum().item()
    
    return running_loss / len(loader), correct / total

def evaluate(model, loader, criterion, device):
    model.eval()
    running_loss = 0.0

            running_loss += loss.item()
            _, predicted = outputs.max(1)
            total += labels.size(0)
            correct += predicted.eq(labels).sum().item()
    
    return running_loss / len(loader), correct / total

🚀 Method 3: Transfer Learning (Pre-trained Models)

🧠 What is Transfer Learning?

Instead of training from scratch, use a model already trained on millions of images (ImageNet: 1.2M images, 1000 classes). The pre-trained model has already learned universal features (edges, textures, shapes). We just add our own classification head on top.

Analogy: Like hiring an experienced artist and just teaching them what labels to write — they already know how to see.

Pre-trained Model (e.g., ResNet50 trained on ImageNet) ┌───────────────────────────────────────────────────────┐ │ Feature Extractor (FROZEN - don't train) │ │ ┌─────────┐ ┌─────────┐ ┌─────────┐ │ │ │ Conv │→ │ Conv │→ │ Conv │→ ... │ │ │ Blocks │ │ Blocks │ │ Blocks │ │ │ │(edges) │ │(texture)│ │(objects)│ │ │ └─────────┘ └─────────┘ └─────────┘ │ └──────────────────────────────────┬────────────────────┘ │ Features (2048-dim vector) ▼ ┌───────────────────────────────────────────────────────┐ │ New Classification Head (TRAINABLE) │ │ Dense(256) → ReLU → Dropout → Dense(num_classes) │ │ → Softmax │ └───────────────────────────────────────────────────────┘ │ ▼ Prediction: "Cat" 97%

🎯 Method 4: Fine-Tuning Pre-trained Models

Fine-Tuning vs Feature Extraction

Feature Extraction (Method 3): Freeze entire base model, only train new head. Fast but limited.

Fine-Tuning (Method 4): After initial training, UNFREEZE some top layers of base model and train them with a very low learning rate. This adapts the pre-trained features to your specific task.

Fine-Tuning Strategy (2-Phase Training): Phase 1: Feature Extraction (5-10 epochs) ┌─────────────────────────────────────┐ │ Base Model Layers: ❄️ FROZEN │ │ New Head Layers: 🔥 TRAINING │ │ Learning Rate: 0.001 │ └─────────────────────────────────────┘ Phase 2: Fine-Tuning (10-20 epochs) ┌─────────────────────────────────────┐ │ Early Layers: ❄️ FROZEN │ ← Generic features (keep) │ Top ~30% Layers: 🔥 TRAINING │ ← Task-specific features (adapt) │ New Head Layers: 🔥 TRAINING │ │ Learning Rate: 0.0001 (10x lower) │ └─────────────────────────────────────┘

def fine_tune_resnet(num_classes=3, device='cuda'):

        nn.ReLU(),
        nn.Dropout(0.5),
        nn.Linear(256, num_classes)

    optimizer = optim.Adam(model.fc.parameters(), lr=1e-3)
    criterion = nn.CrossEntropyLoss()
    
    print("Phase 1: Training classifier head...")

    # Unfreeze layer4 (last residual block) + fc
    for param in model.layer4.parameters():
        param.requires_grad = True

    scheduler = optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=20)
    
    print("\nPhase 2: Fine-tuning top layers...")
    best_acc = 0
    for epoch in range(20):
        train_loss, train_acc = train_one_epoch(model, train_loader, criterion, optimizer, device)
        val_loss, val_acc = evaluate(model, val_loader, criterion, device)
        scheduler.step()
        
        if val_acc > best_acc:

        print(f"  Epoch {epoch+1}: Train Acc={train_acc:.4f}, Val Acc={val_acc:.4f}")

    return model

🔮 Method 5: Vision Transformers (ViT)

Vision Transformers - The New Era

ViT applies the Transformer architecture (originally for NLP) to images. Instead of processing pixels through convolutions, it:

  1. Splits image into fixed-size patches (e.g., 16×16 pixels)
  2. Flattens each patch into a vector
  3. Adds position embeddings (so model knows where patches are)
  4. Processes through Transformer encoder (self-attention)
  5. Uses [CLS] token output for classification
Input Image (224×224) │ ▼ Split into 14×14 = 196 patches (each 16×16 pixels) ┌──────────────────────────────────────────────────────────┐ │ [CLS] [P1] [P2] [P3] ... [P196] ← Patch Embeddings │ │ + + + + + │ │ [E0] [E1] [E2] [E3] ... [E196] ← Position Embeddings │ ├──────────────────────────────────────────────────────────┤ │ Transformer Encoder × 12 layers │ │ ┌─────────────────────────────────────┐ │ │ │ Multi-Head Self-Attention │ │ │ │ Feed-Forward Network │ │ │ │ Layer Normalization │ │ │ └─────────────────────────────────────┘ │ ├──────────────────────────────────────────────────────────┤ │ [CLS] output → MLP Head → num_classes │ └──────────────────────────────────────────────────────────┘ │ ▼ Prediction: "Dog" 96%

11. Evaluation Metrics & Accuracy Comparison

Understanding Metrics

📊 Key Metrics Explained

  • Accuracy: % of correct predictions. Can be misleading with imbalanced classes.
  • Precision: Of all predicted "cat", how many were actually cat? (Low false positives)
  • Recall: Of all actual cats, how many did we find? (Low false negatives)
  • F1-Score: Harmonic mean of precision & recall. Best single metric.
  • Confusion Matrix: Shows exact breakdown of predictions vs truth
  • AUC-ROC: Area under ROC curve. Measures classification ability across thresholds.

12. Key Concepts Deep Dive

🔄 Overfitting vs Underfitting

Overfitting: Model memorizes training data but fails on new data. Signs: high train accuracy, low val accuracy.

Underfitting: Model too simple to learn patterns. Signs: low accuracy on both train and val.

Solutions for Overfitting:

  • More training data / data augmentation
  • Dropout (randomly zero out neurons during training)
  • L2 Regularization (penalize large weights)
  • Early Stopping (stop when val loss increases)
  • Simpler model architecture

📐 Batch Size & Learning Rate

Batch Size: Number of images processed before updating weights.

  • Small batch (8-16): More noise in updates, better generalization, slower training
  • Large batch (64-128): Smoother updates, faster training, may generalize worse
  • Sweet spot: 32 for most cases

Learning Rate: How big a step to take when updating weights.

  • Too high: Training diverges (loss goes up)
  • Too low: Training takes forever, may get stuck
  • Start at 0.001, use ReduceLROnPlateau to decrease automatically

🏗️ Batch Normalization

Normalizes the output of each layer to have zero mean and unit variance. Benefits:

  • Faster training (allows higher learning rates)
  • Acts as regularization
  • Reduces internal covariate shift
  • Place AFTER Conv/Dense, BEFORE activation

🎲 Dropout

During training, randomly sets a fraction of neurons to 0. This prevents co-adaptation (neurons relying too much on each other).

  • Typical rates: 0.25 after conv blocks, 0.5 after dense layers
  • Only active during training (not inference)
  • Acts like training many smaller networks and averaging

🎯 Global Average Pooling vs Flatten

Flatten: Converts 7×7×512 → 25,088 neurons. Many parameters, prone to overfitting.

Global Average Pooling: Averages each feature map to single value. 7×7×512 → 512 neurons. Far fewer parameters, better generalization. Always prefer GAP.

📊 Cross-Entropy Loss

The standard loss function for classification:

  • Binary Cross-Entropy: For 2 classes: -[y·log(p) + (1-y)·log(1-p)]
  • Categorical Cross-Entropy: For N classes: -Σ y_i · log(p_i)
  • Heavily penalizes confident wrong predictions
  • Use sparse_categorical_crossentropy if labels are integers (not one-hot)

⚡ Learning Rate Schedules

Changing learning rate during training improves results:

  • Step Decay: Reduce LR by factor every N epochs
  • Cosine Annealing: Smoothly decreases LR following cosine curve
  • ReduceLROnPlateau: Reduce when validation loss stops improving (most practical)
  • Warmup: Start with very low LR, gradually increase, then decay
  • One Cycle: LR goes up then down in one cycle (fast convergence)

🧊 Catastrophic Forgetting

When fine-tuning, if learning rate is too high, the model "forgets" what it learned from ImageNet. Solutions:

  • Use very low learning rate for pre-trained layers (1e-5)
  • Differential learning rates (lower for early layers)
  • Gradual unfreezing (unfreeze layers one at a time)
  • Freeze BatchNorm layers during fine-tuning

def predict_image_pytorch(model, image_path, class_names, device='cuda'):
    """Predict single image with PyTorch model."""
    from torchvision import transforms
    
    transform = transforms.Compose([
        transforms.Resize((224, 224)),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
    ])
    
    img = Image.open(image_path).convert('RGB')
    img_tensor = transform(img).unsqueeze(0).to(device)  # (1, 3, 224, 224)
    
    model.eval()
    with torch.no_grad():
        outputs = model(img_tensor)
        probabilities = torch.softmax(outputs, dim=1)[0]
        predicted_idx = probabilities.argmax().item()
        confidence = probabilities[predicted_idx].item()
    
    print(f"Predicted: {class_names[predicted_idx]} ({confidence*100:.1f}%)")
    return class_names[predicted_idx], confidence

def extract_features(image_path):
    """Extract HOG + Color Histogram features from image."""
    img = cv2.imread(image_path)
    if img is None:
        return None
    img = cv2.resize(img, (128, 128))
    
    # HOG features
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    hog_feat = hog(gray, orientations=9, pixels_per_cell=(8,8),
                   cells_per_block=(2,2), feature_vector=True)
    
    # Color histogram
    color_feat = []
    for ch in range(3):
        hist = cv2.calcHist([img], [ch], None, [32], [0, 256]).flatten()
        color_feat.extend(hist)
    
    return np.concatenate([hog_feat, np.array(color_feat)])

def load_features(split_dir):
    """Load all images from a split and extract features."""
    X, y = [], []

        class_dir = os.path.join(split_dir, class_name)

            continue
        for img_file in os.listdir(class_dir):
            feat = extract_features(os.path.join(class_dir, img_file))
            if feat is not None:
                X.append(feat)
                y.append(label_idx)
    return np.array(X), np.array(y)
