Put everything together. Build a complete, production-ready AI system from scratch that combines RAG, agents, tool calling, memory, and deployment. This is your portfolio piece.

The capstone = your final exam + job application combined. It's not about getting 100% on theory — it's about proving you can build a real thing that works, end to end. Employers and clients care about what you've BUILT, not what you've read.

The Project: AI Research Assistant

Build a complete AI Research Assistant that can:

  • Chat with users naturally (streaming responses)
  • Search and retrieve information from documents (RAG)
  • Remember user preferences across sessions (memory)
  • Use tools to search the web, run code, and process files
  • Route between different specialized modes
  • Be deployed as a Docker container with an API

Implementation Guide

Phase 1: Core Chat Engine

# core/chat_engine.py — The heart of your assistant
from openai import OpenAI
from typing import Generator

class ChatEngine:
    def __init__(self, model: str = "llama3.1:8b"):
        self.client = OpenAI(base_url="http://localhost:11434/v1", api_key="local")
        self.model = model
        self.system_prompt = "You are a helpful AI research assistant."
    
    def chat(self, message: str, history: list = None, context: str = "") -> str:
        messages = self._build_messages(message, history, context)
        response = self.client.chat.completions.create(
            model=self.model, messages=messages, temperature=0.7
        )
        return response.choices[0].message.content
    
    def chat_stream(self, message: str, history: list = None, context: str = "") -> Generator:
        messages = self._build_messages(message, history, context)
        stream = self.client.chat.completions.create(
            model=self.model, messages=messages, temperature=0.7, stream=True
        )
        for chunk in stream:
            if chunk.choices[0].delta.content:
                yield chunk.choices[0].delta.content
    
    def _build_messages(self, message, history, context):
        system = self.system_prompt
        if context:
            system += f"\n\nRelevant context:\n{context}"
        messages = [{"role": "system", "content": system}]
        if history:
            for user_msg, bot_msg in history[-10:]:
                messages.append({"role": "user", "content": user_msg})
                messages.append({"role": "assistant", "content": bot_msg})
        messages.append({"role": "user", "content": message})
        return messages

Phase 2: RAG Pipeline

# core/rag.py — Document search and retrieval
import chromadb
from sentence_transformers import SentenceTransformer

class RAGPipeline:
    def __init__(self):
        self.embedder = SentenceTransformer("all-MiniLM-L6-v2")
        self.chroma = chromadb.PersistentClient(path="./data/vectordb")
        self.collection = self.chroma.get_or_create_collection("documents")
    
    def add_document(self, text: str, source: str):
        """Chunk and index a document."""
        chunks = self._chunk_text(text, chunk_size=500, overlap=50)
        for i, chunk in enumerate(chunks):
            self.collection.add(
                ids=[f"{source}_{i}"],
                documents=[chunk],
                metadatas=[{"source": source, "chunk": i}]
            )
    
    def search(self, query: str, top_k: int = 3) -> str:
        """Search for relevant context."""
        if self.collection.count() == 0:
            return ""
        results = self.collection.query(query_texts=[query], n_results=top_k)
        return "\n\n".join(results["documents"][0])
    
    def _chunk_text(self, text, chunk_size, overlap):
        words = text.split()
        chunks = []
        for i in range(0, len(words), chunk_size - overlap):
            chunks.append(" ".join(words[i:i + chunk_size]))
        return chunks

Phase 3: Memory + Tools + API

# app.py — FastAPI application tying it all together
from fastapi import FastAPI, UploadFile, File
from fastapi.responses import StreamingResponse
from pydantic import BaseModel

app = FastAPI(title="AI Research Assistant")

# Initialize components
chat_engine = ChatEngine()
rag = RAGPipeline()

class ChatRequest(BaseModel):
    message: str
    user_id: str = "default"

@app.post("/chat")
async def chat_endpoint(request: ChatRequest):
    # 1. Search for relevant context
    context = rag.search(request.message)
    # 2. Generate response with context
    response = chat_engine.chat(request.message, context=context)
    return {"response": response}

@app.post("/upload")
async def upload_document(file: UploadFile = File(...)):
    content = (await file.read()).decode("utf-8")
    rag.add_document(content, source=file.filename)
    return {"status": "indexed", "filename": file.filename}

@app.get("/health")
async def health():
    return {"status": "healthy", "documents": rag.collection.count()}
