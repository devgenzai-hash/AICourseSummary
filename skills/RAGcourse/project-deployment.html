@app.post("/query", response_model=QueryResponse)
async def query(request: QueryRequest):
    """
    Query the RAG pipeline.
    Returns an answer with citations, confidence, and trace ID.
    """
    start = time.time()
    metrics.query_count.inc()
    
    with trace_span("rag_query") as span:
        span.set_attribute("query.text", request.question)
        span.set_attribute("query.source_filter", request.source_filter or "none")
        
        # Check cache first
        if not request.skip_cache:
            cached_response = cache.get(request.question)
            if cached_response:
                metrics.cache_hits.inc()
                span.set_attribute("cache.hit", True)
                cached_response["cached"] = True
                cached_response["latency_ms"] = (time.time() - start) * 1000
                return QueryResponse(**cached_response)
        
        # Execute full pipeline
        result = rag_pipeline.query(
            question=request.question,
            conversation=request.conversation,
            source_filter=request.source_filter
        )
        
        latency = (time.time() - start) * 1000
        metrics.query_latency.observe(latency)
        span.set_attribute("query.latency_ms", latency)
        span.set_attribute("query.decision", result.decision)
        
        response = QueryResponse(
            answer=result.answer,
            citations=result.citations,
            confidence=result.confidence,
            decision=result.decision,
            sources=result.sources,
            trace_id=result.trace_id,
            latency_ms=latency,
            cached=False
        )
        
        # Cache the response
        cache.set(request.question, response.model_dump())
        
        return response

@app.post("/ingest", response_model=IngestResponse)
async def ingest(request: IngestRequest):
    """Ingest documents from a directory into the RAG pipeline."""
    with trace_span("ingest") as span:
        span.set_attribute("ingest.directory", request.directory)
        
        stats = ingestion_pipeline.ingest_directory(request.directory)
        metrics.docs_ingested.inc(stats["documents_unique"])
        
        return IngestResponse(**stats)

2. Redis Semantic Cache

# src/api/cache.py

import json
import hashlib
import numpy as np
import redis
from typing import Optional
from src.ingestion.embedder import EmbeddingGenerator
from src.config import settings

class SemanticCache:
    """
    Two-layer cache using Redis:
    
    Layer 1: Exact match (SHA256 hash of normalized query)
    Layer 2: Semantic match (cosine similarity of query embeddings)
    
    TTL: 1 hour for exact, 30 min for semantic (less confident match)
    """
    
    def __init__(self):
        self.redis = redis.Redis(
            host=settings.redis_host,
            port=settings.redis_port,
            decode_responses=True
        )
        self.embedder = EmbeddingGenerator()
        self.similarity_threshold = 0.95  # Very high threshold for cache hits
        self.exact_ttl = 3600       # 1 hour
        self.semantic_ttl = 1800    # 30 minutes
    
    def get(self, query: str) -> Optional[dict]:
        """Try to get a cached response. Exact match first, then semantic."""
        # Layer 1: Exact match
        query_hash = self._hash_query(query)
        cached = self.redis.get(f"exact:{query_hash}")
        if cached:
            return json.loads(cached)
        
        # Layer 2: Semantic match
        query_embedding = self.embedder.embed_query(query)
        
        # Check against recent query embeddings
        semantic_keys = self.redis.keys("semantic:*")
        for key in semantic_keys[:100]:  # Limit to 100 comparisons
            stored = self.redis.hgetall(key)
            if not stored:
                continue
            
            stored_embedding = np.frombuffer(
                bytes.fromhex(stored["embedding"]), dtype=np.float32
            )
            
            # Cosine similarity
            similarity = np.dot(query_embedding, stored_embedding)
            
            if similarity >= self.similarity_threshold:
                return json.loads(stored["response"])
        
        return None
    
    def set(self, query: str, response: dict):
        """Cache a response for both exact and semantic retrieval."""
        query_hash = self._hash_query(query)
        response_json = json.dumps(response, default=str)
        
        # Layer 1: Exact match cache
        self.redis.setex(f"exact:{query_hash}", self.exact_ttl, response_json)
        
        # Layer 2: Semantic cache (store embedding + response)
        query_embedding = self.embedder.embed_query(query)
        self.redis.hset(f"semantic:{query_hash}", mapping={
            "embedding": query_embedding.astype(np.float32).tobytes().hex(),
            "response": response_json,
            "query": query
        })
        self.redis.expire(f"semantic:{query_hash}", self.semantic_ttl)
    
    def _hash_query(self, query: str) -> str:
        """Normalize and hash a query for exact matching."""
        normalized = query.lower().strip()
        return hashlib.sha256(normalized.encode()).hexdigest()[:16]

3. OpenTelemetry Tracing

# src/api/tracing.py

from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.jaeger.thrift import JaegerExporter
from opentelemetry.instrumentation.fastapi import FastAPIInstrumentor
from opentelemetry.instrumentation.httpx import HTTPXClientInstrumentor
from contextlib import contextmanager
from src.config import settings

def setup_tracing(app):
    """Configure OpenTelemetry with Jaeger exporter."""
    
    # Create tracer provider
    provider = TracerProvider()
    
    # Jaeger exporter (sends traces to Jaeger)
    jaeger_exporter = JaegerExporter(
        agent_host_name=settings.jaeger_host,
        agent_port=settings.jaeger_port,
    )
    
    # Batch processor (buffers spans for efficiency)
    provider.add_span_processor(
        BatchSpanProcessor(jaeger_exporter)
    )
    
    # Set as global provider
    trace.set_tracer_provider(provider)
    
    # Auto-instrument FastAPI (creates spans for all endpoints)
    FastAPIInstrumentor.instrument_app(app)
    
    # Auto-instrument HTTP clients (OpenAI, Cohere calls)
    HTTPXClientInstrumentor().instrument()

@contextmanager
def trace_span(name: str):
    """Create a tracing span for a pipeline stage."""
    tracer = trace.get_tracer("enterprise-rag")
    with tracer.start_as_current_span(name) as span:
        yield span

5. RAGAS Evaluation Pipeline

# src/evaluation/ragas_eval.py

from ragas import evaluate
from ragas.metrics import (
    faithfulness,
    answer_relevancy,
    context_precision,
    context_recall
)
from datasets import Dataset
from typing import List, Dict
import json
from datetime import datetime

class RAGASEvaluator:
    """
    Continuous evaluation using RAGAS framework.
    
    Metrics:
    - Faithfulness: Are answers grounded in context? (anti-hallucination)
    - Answer Relevancy: Do answers address the question?
    - Context Precision: Are retrieved contexts relevant?
    - Context Recall: Did we find all relevant contexts?
    
    Run daily on a test set + weekly on adversarial queries.
    """
    
    def evaluate_batch(self, test_cases: List[Dict]) -> Dict:
        """
        Evaluate a batch of test cases.
        
        Each test case: {
            "question": str,
            "answer": str (generated),
            "contexts": List[str] (retrieved chunks),
            "ground_truth": str (expected answer)
        }
        """
        dataset = Dataset.from_dict({
            "question": [t["question"] for t in test_cases],
            "answer": [t["answer"] for t in test_cases],
            "contexts": [t["contexts"] for t in test_cases],
            "ground_truth": [t["ground_truth"] for t in test_cases],
        })
        
        # Run RAGAS evaluation
        results = evaluate(
            dataset,
            metrics=[
                faithfulness,
                answer_relevancy,
                context_precision,
                context_recall
            ]
        )
        
        # Format results
        report = {
            "timestamp": datetime.now().isoformat(),
            "num_cases": len(test_cases),
            "scores": {
                "faithfulness": float(results["faithfulness"]),
                "answer_relevancy": float(results["answer_relevancy"]),
                "context_precision": float(results["context_precision"]),
                "context_recall": float(results["context_recall"])
            },
            "pass": all(v >= 0.8 for v in results.values())
        }
        
        # Save report
        with open(f"evals/report_{datetime.now().strftime('%Y%m%d')}.json", "w") as f:
            json.dump(report, f, indent=2)
        
        return report

# alerts.yml

groups:
  - name: rag_alerts
    rules:
      # High latency alert
      - alert: HighQueryLatency
        expr: histogram_quantile(0.95, rag_query_latency_ms) > 2000
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "RAG query P95 latency > 2s"
      
      # Hallucination rate alert
      - alert: HighHallucinationRate
        expr: rate(rag_hallucinations_total[1h]) / rate(rag_queries_total[1h]) > 0.05
        for: 10m
        labels:
          severity: critical
        annotations:
          summary: "Hallucination rate > 5%"
      
      # Low faithfulness score
      - alert: LowFaithfulness
        expr: rag_ragas_faithfulness < 0.8
        for: 30m
        labels:
          severity: critical
        annotations:
          summary: "RAGAS faithfulness score dropped below 0.8"

      - alert: ServiceDown
        expr: up{job="rag-api"} == 0
        for: 1m
        labels:
          severity: critical
