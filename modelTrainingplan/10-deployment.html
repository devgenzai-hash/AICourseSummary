Chapter 10: Deployment

From trained adapter to production API: LoRA merging, quantization (GGUF/AWQ/GPTQ), serving with vLLM and Text Generation Inference, and building a REST API.

🔗 Step 1: Merge LoRA Adapters

"""
Merge all LoRA adapters into the base model to create a standalone model.
This eliminates the need to load adapters at inference time.

Pipeline: Base Model + CPT LoRA + SFT LoRA + DPO LoRA → Merged Model
"""
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM
from peft import PeftModel

BASE_MODEL = "google/gemma-3-27b-it"
ADAPTERS = [
    "./checkpoints/cpt_lora/final",   # Stage 1: CPT
    "./checkpoints/sft_lora/final",   # Stage 2: SFT
    "./checkpoints/dpo_lora/final",   # Stage 3: DPO
]
OUTPUT_DIR = "./model_merged"

print("📦 Loading base model in full precision...")
model = AutoModelForCausalLM.from_pretrained(
    BASE_MODEL,
    torch_dtype=torch.bfloat16,
    device_map="cpu",  # Load on CPU for merging (needs ~60GB RAM for 27B)
)
tokenizer = AutoTokenizer.from_pretrained(BASE_MODEL)

# Merge each adapter sequentially
for adapter_path in ADAPTERS:
    print(f"  📎 Merging adapter: {adapter_path}")
    model = PeftModel.from_pretrained(model, adapter_path)
    model = model.merge_and_unload()

# Save merged model
print(f"💾 Saving merged model to {OUTPUT_DIR}...")
model.save_pretrained(OUTPUT_DIR, safe_serialization=True)
tokenizer.save_pretrained(OUTPUT_DIR)

print(f"✅ Merged model saved! Size: {sum(p.numel() for p in model.parameters()) / 1e9:.1f}B params")

📦 Step 2: Quantization for Deployment

Option A: GGUF (for llama.cpp / Ollama)

# Convert to GGUF format for CPU/consumer GPU inference
# Requires: llama.cpp (clone from GitHub)

# 1. Install llama.cpp
git clone https://github.com/ggerganov/llama.cpp
cd llama.cpp && make -j

# 2. Convert HuggingFace model to GGUF
python convert_hf_to_gguf.py ../model_merged \
    --outfile ../model_merged/gemma-scripture-f16.gguf \
    --outtype f16

# 3. Quantize to different sizes
# Q4_K_M: Best quality/size balance (recommended)
./llama-quantize ../model_merged/gemma-scripture-f16.gguf \
    ../model_merged/gemma-scripture-Q4_K_M.gguf Q4_K_M

# Q5_K_M: Higher quality, larger size
./llama-quantize ../model_merged/gemma-scripture-f16.gguf \
    ../model_merged/gemma-scripture-Q5_K_M.gguf Q5_K_M

# Q8_0: Near full precision
./llama-quantize ../model_merged/gemma-scripture-f16.gguf \
    ../model_merged/gemma-scripture-Q8_0.gguf Q8_0

Option B: AWQ (for vLLM — fastest inference)

"""
AWQ (Activation-Aware Weight Quantization)
Best for GPU inference with vLLM. 4-bit with minimal quality loss.
"""
from awq import AutoAWQForCausalLM
from transformers import AutoTokenizer

MODEL_PATH = "./model_merged"
QUANT_PATH = "./model_awq"

print("📦 Loading model for AWQ quantization...")
model = AutoAWQForCausalLM.from_pretrained(MODEL_PATH, device_map="auto")
tokenizer = AutoTokenizer.from_pretrained(MODEL_PATH)

# Calibration data — use a sample of your training data
calibration_data = [
    "Explain the concept of Dharma in the Bhagavad Gita.",
    "कर्मयोग क्या है और इसका महत्व क्या है?",
    "What are the four paths of Yoga according to Hindu philosophy?",
    "Describe the Vishvarupa (universal form) shown by Krishna.",
    # Add 100-500 diverse samples for best calibration
]

# Quantize
quant_config = {
    "zero_point": True,
    "q_group_size": 128,
    "w_bit": 4,
    "version": "GEMM"
}

print("⚡ Quantizing with AWQ...")
model.quantize(tokenizer, quant_config=quant_config, calib_data=calibration_data)

# Save
model.save_quantized(QUANT_PATH)
tokenizer.save_pretrained(QUANT_PATH)
print(f"✅ AWQ model saved to {QUANT_PATH}")

"""
Python client for the vLLM server (OpenAI-compatible).
"""
import openai

# Point to your vLLM server
client = openai.OpenAI(
    base_url="http://localhost:8000/v1",
    api_key="not-needed",  # vLLM doesn't require auth by default
)

def ask_scripture_model(question: str, language: str = "en") -> str:
    """Query the deployed scripture model."""
    
    system_prompt = (
        "You are a Hindu scripture scholar. Cite specific references."
        if language == "en" else
        "आप हिंदू शास्त्रों के विद्वान हैं। विशिष्ट संदर्भ दें।"
    )
    
    response = client.chat.completions.create(
        model="./model_awq",  # Model name as registered in vLLM
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": question}
        ],
        temperature=0.7,
        max_tokens=1024,
        stream=True,  # Streaming for real-time responses
    )
    
    full_response = ""
    for chunk in response:
        if chunk.choices[0].delta.content:
            content = chunk.choices[0].delta.content
            full_response += content
            print(content, end="", flush=True)
    
    print()
    return full_response

# Usage
answer = ask_scripture_model("What is the meaning of Bhagavad Gita 2.47?")
answer_hi = ask_scripture_model("गीता में कर्मयोग का क्या अर्थ है?", language="hi")

{{ .System }}
{{ end }}user
{{ .Prompt }}
model
{{ .Response }}"""

🌐 Production REST API

"""
Production-ready FastAPI wrapper for the scripture model.
Features: rate limiting, request validation, health checks, metrics.
"""
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field
import openai
import time
import uvicorn

app = FastAPI(
    title="Hindu Scripture AI API",
    description="AI-powered Hindu scripture knowledge assistant",
    version="1.0.0",
)

# Connect to vLLM backend
vllm_client = openai.OpenAI(
    base_url="http://localhost:8000/v1",
    api_key="not-needed",
)

# ============================================
# Request/Response Models
# ============================================
class QuestionRequest(BaseModel):
    question: str = Field(..., min_length=5, max_length=2000)
    language: str = Field(default="en", pattern="^(en|hi|sa)$")
    max_tokens: int = Field(default=1024, ge=50, le=4096)
    temperature: float = Field(default=0.7, ge=0.0, le=1.5)
    include_references: bool = Field(default=True)

class QuestionResponse(BaseModel):
    answer: str
    language: str
    tokens_used: int
    latency_ms: float


# ============================================
# Endpoints
# ============================================
@app.get("/health")
async def health():
    return {"status": "healthy", "model": "gemma-scripture-27b"}


@app.post("/ask", response_model=QuestionResponse)
async def ask_question(request: QuestionRequest):
    """Ask a question about Hindu scriptures."""
    
    start_time = time.time()
    
    system_prompts = {
        "en": "You are a Hindu scripture scholar. Cite specific verse references.",
        "hi": "आप हिंदू शास्त्रों के विद्वान हैं। विशिष्ट श्लोक संदर्भ दें।",
        "sa": "त्वं हिन्दूशास्त्राणां विद्वान् असि। विशिष्टान् श्लोकसन्दर्भान् प्रयच्छ।",
    }
    
    try:
        response = vllm_client.chat.completions.create(
            model="gemma-scripture",
            messages=[
                {"role": "system", "content": system_prompts[request.language]},
                {"role": "user", "content": request.question}
            ],
            temperature=request.temperature,
            max_tokens=request.max_tokens,
        )
        
        answer = response.choices[0].message.content
        tokens = response.usage.total_tokens
        
    except Exception as e:
        raise HTTPException(status_code=503, detail=f"Model unavailable: {str(e)}")
    
    latency = (time.time() - start_time) * 1000
    
    return QuestionResponse(
        answer=answer,
        language=request.language,
        tokens_used=tokens,
        latency_ms=round(latency, 1),
    )


if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8080)
