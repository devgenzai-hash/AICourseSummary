🧮 Memory Optimization Techniques

"""
Memory optimization strategies — ranked by effectiveness.
Apply from top to bottom until your model fits in memory.
"""

# ============================================
# Level 1: QLoRA (reduces model memory by ~4x)
# ============================================
# Already covered in training chapters
# Base: 27B × 2 bytes (BF16) = 54 GB
# QLoRA: 27B × 0.5 bytes (NF4) + LoRA params = ~14 GB

# ============================================
# Level 2: Gradient Checkpointing (reduces activation memory by ~60%)
# ============================================
# In training args:
# gradient_checkpointing=True
# Trades compute (20-30% slower) for memory

# ============================================
# Level 3: Reduce Batch Size + Increase Gradient Accumulation
# ============================================
# Memory ∝ batch_size × seq_length²
# Instead of batch=8, use batch=1 with grad_accum=8
# Same effective batch, 8x less memory

# ============================================
# Level 4: Reduce Sequence Length
# ============================================
# Memory scales with seq_length² (attention)
# 4096 tokens = 4× the memory of 2048 tokens
# Use dynamic padding + shorter sequences during early training

# ============================================
# Level 5: Flash Attention 2 (reduces attention memory)
# ============================================
# attn_implementation="flash_attention_2"  # in model loading
# Reduces O(n²) memory to O(n) for attention
# FREE performance gain — always use it

# ============================================
# Level 6: CPU Offloading (DeepSpeed ZeRO-Offload)
# ============================================
# Moves optimizer states and/or parameters to CPU RAM
# Massive memory savings at cost of speed (30-50% slower)
# Good if you have lots of CPU RAM (>128GB)

# ============================================
# Level 7: Smaller LoRA Rank
# ============================================
# r=64 → r=16: reduces trainable params by 4×
# Each LoRA layer: 2 × hidden_dim × r parameters
# For 27B model: r=64 ~= 200M params, r=16 ~= 50M params

# ============================================
# MEMORY ESTIMATION FORMULA
# ============================================
def estimate_memory_gb(
    model_params_B: float,  # in billions
    precision: str = "4bit",  # "4bit", "8bit", "bf16", "fp32"
    lora_rank: int = 32,
    batch_size: int = 2,
    seq_length: int = 2048,
    gradient_checkpointing: bool = True,
) -> dict:
    """Estimate GPU memory requirements."""
    
    # Model weights
    bytes_per_param = {
        "4bit": 0.5, "8bit": 1.0, "bf16": 2.0, "fp32": 4.0
    }[precision]
    model_memory = model_params_B * bytes_per_param
    
    # LoRA parameters (in BF16)
    # Rough: 2 × hidden_dim × rank × num_layers × num_target_modules
    # For 27B: hidden=5120, layers=46, targets=7
    hidden_dim = int(model_params_B * 190)  # rough scaling
    num_layers = int(model_params_B * 1.7)
    lora_params = 2 * hidden_dim * lora_rank * num_layers * 7
    lora_memory = lora_params * 2 / 1e9  # BF16
    
    # Optimizer states (AdamW: 2× params for momentum + variance)
    optimizer_memory = lora_memory * 2  # Only for LoRA params
    
    # Activations (rough estimate)
    activation_factor = 0.4 if gradient_checkpointing else 1.0
    activation_memory = (
        batch_size * seq_length * hidden_dim * num_layers * 2 / 1e9
        * activation_factor
    )
    
    total = model_memory + lora_memory + optimizer_memory + activation_memory
    
    return {
        "model_weights_gb": round(model_memory, 1),
        "lora_params_gb": round(lora_memory, 1),
        "optimizer_gb": round(optimizer_memory, 1),
        "activations_gb": round(activation_memory, 1),
        "total_estimated_gb": round(total * 1.1, 1),  # 10% overhead
        "recommended_gpu": (
            "RTX 4090 24GB" if total < 20 else
            "A100 40GB" if total < 35 else
            "A100 80GB" if total < 70 else
            f"{int(total/40)+1}× A100 40GB"
        )
    }

# Example usage:
print(estimate_memory_gb(27, "4bit", lora_rank=32, batch_size=2, seq_length=4096))
# {'model_weights_gb': 13.5, 'lora_params_gb': 1.2, 'optimizer_gb': 2.4,
#  'activations_gb': 4.8, 'total_estimated_gb': 24.1, 'recommended_gpu': 'A100 40GB'}

📜 Environment Setup Scripts

# ============================================
# setup.sh — Complete environment setup
# Run on fresh GPU instance (Ubuntu 22.04 + CUDA 12.1+)
# ============================================

#!/bin/bash
set -e

echo "🔧 Setting up Gemma fine-tuning environment..."

# 1. System packages
sudo apt-get update && sudo apt-get install -y \
    python3.11 python3.11-venv python3-pip \
    git git-lfs htop nvtop tmux

# 2. Create virtual environment
python3.11 -m venv ~/gemma-env
source ~/gemma-env/bin/activate

# 3. Install PyTorch (CUDA 12.1)
pip install torch==2.3.0 torchvision torchaudio \
    --index-url https://download.pytorch.org/whl/cu121

# 4. Install training stack
pip install \
    transformers>=4.50.0 \
    trl>=0.15.0 \
    peft>=0.14.0 \
    bitsandbytes>=0.43.0 \
    accelerate>=0.30.0 \
    datasets>=2.19.0 \
    wandb \
    flash-attn --no-build-isolation \
    deepspeed>=0.14.0 \
    sentencepiece \
    protobuf

# 5. Login to Hugging Face (for Gemma access)
huggingface-cli login --token $HF_TOKEN

# 6. Login to Weights & Biases
wandb login $WANDB_API_KEY

# 7. Verify installation
python -c "
import torch
print(f'PyTorch: {torch.__version__}')
print(f'CUDA available: {torch.cuda.is_available()}')
print(f'GPU: {torch.cuda.get_device_name(0)}')
print(f'VRAM: {torch.cuda.get_device_properties(0).total_mem / 1e9:.1f} GB')

import transformers, trl, peft, bitsandbytes
print(f'Transformers: {transformers.__version__}')
print(f'TRL: {trl.__version__}')
print(f'PEFT: {peft.__version__}')

echo "✅ Environment ready! Activate with: source ~/gemma-env/bin/activate"
⚠️ Common Setup Issues
  • flash-attn fails to install: Needs CUDA toolkit matching your PyTorch CUDA version. Try pip install flash-attn --no-build-isolation
  • bitsandbytes errors on Windows: Use WSL2 or Linux. BnB has limited Windows support.
  • OOM immediately: Reduce batch_size to 1, enable gradient_checkpointing, reduce seq_length
  • Gemma access denied: Accept Gemma license on HuggingFace model card first
