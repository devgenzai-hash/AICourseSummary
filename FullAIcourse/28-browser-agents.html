Browser Agents

AI agents that can browse the web, click buttons, fill forms, and extract data — like giving your AI robot hands to use a computer. This is the frontier of autonomous AI.

🏠 Daily Life Analogy

Browser agent = a virtual intern sitting at your computer. You tell them "Go to Amazon, search for wireless headphones under $50, and make a comparison table." They open the browser, navigate pages, read content, and bring you back the results.

They can see the page (screenshots or DOM), decide what to click, type in search boxes, and navigate — just like a human would.

Playwright + AI: Basic Browser Agent

# pip install playwright openai
# playwright install chromium

import asyncio
import json
from playwright.async_api import async_playwright
from openai import OpenAI

class BrowserAgent:
    """AI agent that can browse the web and extract information."""
    
    def __init__(self, model: str = "llama3.1:8b"):
        self.client = OpenAI(base_url="http://localhost:11434/v1", api_key="local")
        self.model = model
        self.browser = None
        self.page = None
    
    async def start(self, headless: bool = True):
        """Launch the browser."""
        self.playwright = await async_playwright().start()
        self.browser = await self.playwright.chromium.launch(headless=headless)
        self.page = await self.browser.new_page()
    
    async def stop(self):
        """Close the browser."""
        if self.browser:
            await self.browser.close()
        if self.playwright:
            await self.playwright.stop()
    
    async def get_page_content(self) -> str:
        """Get readable page content (simplified DOM)."""
        # Extract text content and key interactive elements
        content = await self.page.evaluate("""() => {
            const elements = [];
            // Get all text
            const text = document.body.innerText.substring(0, 3000);
            // Get all links
            const links = [...document.querySelectorAll('a[href]')]
                .slice(0, 20)
                .map((a, i) => `[link ${i}] ${a.innerText.trim()} → ${a.href}`);
            // Get all buttons
            const buttons = [...document.querySelectorAll('button, input[type="submit"]')]
                .slice(0, 10)
                .map((b, i) => `[button ${i}] ${b.innerText || b.value}`);
            // Get all inputs
            const inputs = [...document.querySelectorAll('input, textarea, select')]
                .slice(0, 10)
                .map((inp, i) => `[input ${i}] type=${inp.type} name=${inp.name} placeholder=${inp.placeholder}`);
            
            return { text, links: links.join('\\n'), buttons: buttons.join('\\n'), inputs: inputs.join('\\n') };
        }")
        
        return f"""URL: {self.page.url}
Title: {await self.page.title()}

TEXT CONTENT:
{content['text'][:2000]}

LINKS:
{content['links']}

BUTTONS:
{content['buttons']}

INPUTS:
{content['inputs']}"""
    
    async def execute_action(self, action: dict) -> str:
        """Execute a browser action decided by the AI."""
        action_type = action.get("action")
        
        try:
            if action_type == "navigate":
                await self.page.goto(action["url"], wait_until="domcontentloaded")
                return f"Navigated to {action['url']}"
            
            elif action_type == "click":
                await self.page.click(action["selector"])
                await self.page.wait_for_load_state("domcontentloaded")
                return f"Clicked: {action['selector']}"
            
            elif action_type == "type":
                await self.page.fill(action["selector"], action["text"])
                return f"Typed '{action['text']}' into {action['selector']}"
            
            elif action_type == "extract":
                content = await self.get_page_content()
                return content
            
            elif action_type == "scroll":
                await self.page.evaluate("window.scrollBy(0, 500)")
                return "Scrolled down"
            
            elif action_type == "done":
                return "TASK_COMPLETE"
            
            else:
                return f"Unknown action: {action_type}"
        except Exception as e:
            return f"Error: {str(e)}"
    
    async def run_task(self, task: str, max_steps: int = 10) -> str:
        """Run a browsing task autonomously."""
        system = """You are a browser automation agent. You can:
- navigate: Go to a URL
- click: Click an element (use CSS selectors)
- type: Type text into an input field
- extract: Read the current page content
- scroll: Scroll down the page
- done: Task is complete

Respond with a JSON action:
{"action": "navigate", "url": "https://..."}
{"action": "click", "selector": "button.submit"}
{"action": "type", "selector": "input[name='q']", "text": "search query"}
{"action": "extract"}
{"action": "done", "result": "summary of what you found"}"""
        
        messages = [
            {"role": "system", "content": system},
            {"role": "user", "content": f"Task: {task}"}
        ]
        
        for step in range(max_steps):
            print(f"\n--- Step {step + 1} ---")
            
            # Ask AI what to do
            response = self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                temperature=0,
            )
            
            ai_response = response.choices[0].message.content
            messages.append({"role": "assistant", "content": ai_response})
            
            # Parse the action
            try:
                action = json.loads(ai_response)
            except json.JSONDecodeError:
                messages.append({"role": "user", "content": "Invalid JSON. Please respond with a valid JSON action."})
                continue
            
            print(f"Action: {action}")
            
            # Execute the action
            result = await self.execute_action(action)
            
            if result == "TASK_COMPLETE":
                return action.get("result", "Task completed")
            
            print(f"Result: {result[:200]}")
            messages.append({"role": "user", "content": f"Action result:\n{result}"})
        
        return "Max steps reached"

# ====== Usage ======
async def main():
    agent = BrowserAgent()
    await agent.start(headless=False)  # Set True for no visible browser
    
    try:
        result = await agent.run_task(
            "Go to Hacker News, get the titles of the top 5 stories"
        )
        print(f"\n✅ Result: {result}")
    finally:
        await agent.stop()

asyncio.run(main())

AI-Powered Web Scraping Agent

# Intelligent web scraper that adapts to any website
import asyncio
from playwright.async_api import async_playwright
from openai import OpenAI

class SmartScraper:
    """AI-powered scraper that understands page structure."""
    
    def __init__(self):
        self.client = OpenAI(base_url="http://localhost:11434/v1", api_key="local")
    
    async def scrape(self, url: str, instruction: str) -> dict:
        """Scrape a URL based on natural language instructions."""
        async with async_playwright() as p:
            browser = await p.chromium.launch(headless=True)
            page = await browser.new_page()
            await page.goto(url, wait_until="domcontentloaded")
            
            # Get page HTML (simplified)
            html = await page.content()
            text = await page.inner_text("body")
            
            await browser.close()
        
        # Use AI to extract structured data
        prompt = f"""Extract data from this webpage based on the instruction.

Instruction: {instruction}

Page text (first 4000 chars):
{text[:4000]}

Return the extracted data as a JSON object. Use appropriate keys based on the instruction."""
        
        response = self.client.chat.completions.create(
            model="llama3.1:8b",
            messages=[{"role": "user", "content": prompt}],
            temperature=0,
        )
        
        try:
            return json.loads(response.choices[0].message.content)
        except:
            return {"raw": response.choices[0].message.content}

# Usage
scraper = SmartScraper()
data = asyncio.run(scraper.scrape(
    url="https://news.ycombinator.com",
    instruction="Extract the top 5 story titles with their point counts and comment counts"
))
print(json.dumps(data, indent=2))

Anti-Detection & Best Practices

# Making browser agents harder to detect
from playwright.async_api import async_playwright

async def stealth_browser():
    async with async_playwright() as p:
        browser = await p.chromium.launch(
            headless=False,
            args=[
                "--disable-blink-features=AutomationControlled",
                "--no-sandbox",
            ]
        )
        
        context = await browser.new_context(
            viewport={"width": 1920, "height": 1080},
            user_agent="Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
            locale="en-US",
        )
        
        page = await context.new_page()
        
        # Remove automation indicators
        await page.add_init_script("""
            Object.defineProperty(navigator, 'webdriver', { get: () => undefined });
        """)
        
        # Add random delays (human-like behavior)
        import random
        await asyncio.sleep(random.uniform(1, 3))
        
        return browser, page

🏭 Industry Standards

  • Respect robots.txt — Check if the website allows automated access. Scraping prohibited sites is risky.
  • Rate limiting — Add delays between requests (1-3 seconds minimum). Never DDoS a site.
  • browser-use is the hot framework — Open-source, works with any LLM, natural language control.
  • Vision models for complex UIs — When DOM is too messy (dynamic SPAs), screenshot + GPT-4o works better.
  • Headless for production — Run browsers without GUI in containers (Docker + Playwright).
  • Session management — Store cookies/state to avoid re-logging in every time.
  • Error recovery — Pages fail to load, selectors break, popups appear. Build retry logic.

📝 Summary

  • Browser agents = AI + browser automation (Playwright/Puppeteer)
  • AI sees pages via: DOM text, accessibility tree, or screenshots
  • The agent loop: Observe page → Decide action → Execute → Repeat
  • Actions: navigate, click, type, scroll, extract, wait
  • Frameworks: browser-use, Stagehand, Skyvern, custom Playwright
  • Always: rate limit, respect robots.txt, handle errors, add anti-detection

❌ Beginner Mistakes to Avoid

  • No rate limiting — Hammering a site with requests gets you IP-banned instantly.
  • Fragile selectors — Using classes like ".css-1a2b3c" that change every deploy. Use semantic selectors.
  • No error recovery — Pages timeout, elements disappear, popups block. Always wrap in try/except.
  • Ignoring robots.txt — Legal risk. Always check what's allowed before scraping.
  • Sending full page HTML to LLM — Way too many tokens. Extract just the text/relevant elements.
